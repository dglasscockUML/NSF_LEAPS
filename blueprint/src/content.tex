\begin{leanrmk}
    Some basic topology theorems we use over and over:
    \begin{itemize}
        \item \verb|IsClosedMap.closure_image_eq_of_continuous| shows that $\overline{fA} = f\overline{A}$ as long as $f: X \to Y$ is a closed map
        
        \item \verb|IsCompact.image| the continuous image of a compact set is compact

        \item \verb|IsProperMap.isClosedMap| shows that a proper map is closed

        \item \verb|Continuous.isProperMap| shows that a continuous function from a compact space to a T2 space is proper.

        \item \verb|mem_closure_iff|: a point $x$ is in $\overline{A}$ if and only if for all open $U \ni x$, $U \cap \overline{A} \neq \emptyset$.

        \item \verb|IsClosed.isCompact|: a closed subset of a compact space is compact

        \item \verb|IsCompact.isClosed|: in a Hausdorff space, a compact set is closed

        \item \verb|isClosed_closure|: the closure of a set is closed
    \end{itemize}
\end{leanrmk}

Here's a result we use several times that we'll need ready access to in Lean.

\begin{lemma}
\label{lemma_image_closure_equals_closure_image}
 \lean{imageClosureIsClosureImage}
 \leanok{}
    Let $f: X \to Y$ be a continuous map of compact Hausdorff spaces.  For all $A \subseteq X$, $\overline{fA} = f \overline{A}$.
\end{lemma}

\begin{proof}
    By \verb|Continuous.isProperMap|, $f$ is proper. Therefore, by \verb|IsProperMap.isClosedMap|, $f$ is closed. Therefore, by \verb|IsClosedMap.closure_image_eq_of_continuous|, $\overline{fA} = f \overline{A}$.
\end{proof}

\begin{lemma}
    \label{lemma_entourage_regularity_at_a_point}
    \lean{openClosureProductInEntourage}
    \leanok{}
    Let $X$ be a uniform space, $x \in X$, and $\alpha \subseteq X^2$ be an entourage.  There exists an open set $U \ni x$ such that $\overline{U} \times \overline{U} \subseteq \alpha$.
\end{lemma}

\begin{proof}
    Put the following things in order:
    \begin{itemize}
        \item \verb|UniformSpace.ball x \alpha| will give the element of \verb|nbhd x| corresponding to $\alpha$
        
        \item Since $X$ is uniform, by \verb|UniformSpace.to_regularSpace|, it is regular
        
        \item \verb|exists_mem_nhds_isClosed_subset| gives that since $X$ is regular, that there is a closed neighborhood $V$ of $x$ contained in \verb|UniformSpace.ball x \alpha|
    \end{itemize}
\end{proof}


\begin{lemma}[Substitute for semiopenness of a factor map]
    \label{lemma_image_of_entourage_somewhere_dense_along_diagonal}
    \lean{openProductInEntourageImage}
    \leanok{}
    Let $\pi: X \to Y$ be a surjection of compact Hausdorff spaces. For all entourages $\alpha \subseteq X^2$, there exists a non-empty, open set $V \subseteq Y$ such that
    \[V \times V \subseteq (\pi \otimes \pi) \alpha.\]
\end{lemma}

\begin{proof}
\uses{lemma_entourage_regularity_at_a_point}
    Let $\alpha \subseteq X^2$ be an entourage.  For all $x \in X$, the set $\alpha[x]$ is an open neighborhood of $x$.  By \cref{lemma_entourage_regularity_at_a_point}, there exists a non-empty, open set $U_x \subseteq X$ containing $x$ such that $\overline{U_x} \times \overline{U_x} \subseteq \alpha$.

    By compactness, there exists a finite set $F \subseteq X$ such that $\cup_{x \in F} U_x = X$. Thus $\cup_{x \in F} \overline{U_x} = X$.  Since $\pi$ is surjective, $\cup_{x \in F} \pi \overline{U_x} = Y$.  Since $\pi$ is a closed map, the sets $\pi \overline{U_x}$ are closed.  Since $Y$ is compact, Hausdorff, it is a Baire space, so there exists $x \in F$ such that $\pi \overline{U_x}$ has non-empty interior.  Let $V \subseteq Y$ be non-empty, open such that $V \subseteq \pi \overline{U_x}$.  Then
    \[V \times V \subseteq \pi \overline{U_x} \times \pi \overline{U_x} = (\pi \otimes \pi) (\overline{U_x} \times \overline{U_x}) \subseteq (\pi \otimes \pi)\alpha,\]
    as was to be shown.
\end{proof}

\begin{definition}[{Left ideal, \cite[Def. 1.26]{hindman_strauss_book_2012}}]
\lean{SemigroupLeftIdeal}
\leanok{}
    A left ideal of semigroup $S$ is a non-empty subset $L \subseteq S$ such that for all $s \in S$, $s L \subseteq L$.
\end{definition}

\begin{lemma}
    \label{lemma_intersection_of_ideals_is_ideal}
    \lean{nonemptyInterOfLeftIdealsIsLeftIdeal}
    \leanok{}
    Let $S$ be a semigroup, and let $L_i$ be a collection of left ideals of $S$.  The intersection $\bigcap_i L_i$, if non-empty, is a left ideal of $S$.
\end{lemma}

\begin{proof}
\leanok{}
    Suppose $\bigcap_i L_i$ is non-empty.  We need only show that for all $s \in S$, $s\bigcap_i L_i \subseteq \bigcap_i L_i$.  Let $i \in I$.  Since $L_i$ is a left ideal,
    \[s \bigcap_i L_i \subseteq s L_i \subseteq L_i,\]
    as desired.
\end{proof}

\begin{definition}[{Minimal left ideal, \cite[Def. 1.27]{hindman_strauss_book_2012}}]
    \lean{isMinLeftIdeal}
    \leanok{}
    A minimal left ideal of semigroup $S$ is a left ideal $L$ with the property that for all left ideals $J \subseteq S$ with $J \subseteq L$, $J = L$.  (That is, $L$ is a minimal element by containment amongst left ideals.)
\end{definition}

\begin{definition}[Idempotent]
    \lean{IsIdempotentElem}
    \mathlibok{}
    An element $s$ of a semigroup $S$ is idempotent if $ss = s$.
\end{definition}

\subsection{Definitions}

Let $S$ be a set.

\begin{definition}[Collection]
    A \emph{collection} of subsets of $S$ is an element of $\collections{S}$.
\end{definition}

\begin{definition}[Nonempty collection]
    A collection $\family \in \families{S}$ is \emph{nonempty} if $\family \neq \emptyset$.
\end{definition}

\begin{definition}[Proper collection]
    A collection $\family \in \families{S}$ is \emph{proper} if $\family \neq \powerset{S}$.
\end{definition}

\begin{definition}[Upward closed collection]
    A collection of sets $\family \in \collections{S}$ is \emph{upward closed} if for all $A, B \subseteq S$, (($A \in \family$ and $A \subseteq B$) implies $B \in \family$).
\end{definition}

\begin{definition}[Family of subsets of $S$]
    A collection of sets $\family \in \collections{S}$ is a \emph{family} if it is upward closed.  We denote the set of families on $S$ by $\families{S}$.
\end{definition}

\begin{definition}[Partition regular family]
    A family $\family \in \families{S}$ is \emph{partition regular family} if for all $A, B \subseteq S$, $A \cup B \in \family$ implies $A \in \family$ or $B \in \family$.
\end{definition}

\begin{definition}[Filter family]
    A family $\family \in \families{S}$ is a \emph{filter family} if for all $A, B \subseteq S$, $A \in \family$ and $B \in \family$ implies $A \cap B \in \family$.
\end{definition}

\begin{definition}[Ultrafilter family]
    A family $\family \in \families{S}$ is an \emph{ultrafilter family} if it is both a partition regular family and a filter family.
\end{definition}

\subsection{Set lattice}

\begin{enumerate}
    \item $\cup: \families{S}^2 \to \families{S}$
    \begin{enumerate}
        \item The union of two families is a family
        \item The union of two filters is not necessarily a filter
        \item The union of two partition regular families is partition regular
    \end{enumerate}
    \item $\cap : \families{S}^2 \to \families{S}$
        \begin{enumerate}
        \item The intersection of two families is a family
        \item The intersection of two filters is a filter
        \item The intersection of two partition regular families is not necessarily partition regular
    \end{enumerate}
    \item $(\families{S}, \cup, \cap)$ is a sub-lattice of $(\mathcal{P}(\mathcal{P}(S)),\cup,\cap)$
    \item By \cref{thm_de_morgan_union_v1} and \cref{thm_de_morgan_union_v2}, I believe that $(\mathcal{P}(\mathcal{P}(S)),\cup,\cap,*)$ is something like an involutive lattice
\end{enumerate}

\subsection{Dual families}

\begin{definition}[Dual map]
    The \emph{dual} map $*: \mathcal{P}(\mathcal{P}(S)) \to \mathcal{P}(\mathcal{P}(S))$ is defined by 
    \[\family^* = \{A \subseteq S : \forall B \in \family, \ A \cap B \neq \emptyset\}.\]
    For $\family \in \collections{S}$, its image $\family^*$ is the \emph{dual} of $\family$.
\end{definition}

\vskip0.5cm

Facts on the level of collections:
\begin{enumerate}
    \item $*$ is antitone: for collections $\family$ and $\familytwo$, if $\family \subseteq \familytwo$, then $\familytwo^* \subseteq \family^*$
    \begin{itemize}
        \item Proof: \cref{thm_dual_is_antitone}
    \end{itemize}

    \item For all collections $\family$ and $\familytwo$, $(\family \cup \familytwo)^* = \family^* \cap \familytwo^*$ and $(\family \cap \familytwo)^* = \family^* \cup \familytwo^*$
    \begin{itemize}
        \item Proof: \cref{thm_de_morgan_union_v1} and \cref{thm_de_morgan_union_v2}
    \end{itemize}
\end{enumerate}

\vskip0.5cm

Facts on the level of families:
\begin{enumerate}
    \item $*$ maps families to families, that is, $*: \families{S} \to \families{S}$
    \begin{itemize}
        \item Proof: \cref{thm_dual_of_fam_is_fam}
        \item Note that the edge cases $\emptyset^* = \mathcal{P}(S)$ and $\mathcal{P}(S)^* = \emptyset$ hold by strict interpretation of the definition.  Thus, in the theorems proving facts about the dual, there are no separate edge cases to consider.
        \item There is a more general statement here we might favor: $*$ takes $\mathcal{P}(\mathcal{P}(S))$ to $\families{S}$.
    \end{itemize}
    
    \item Useful equivalent formulation: for all families $\family$,
    \[\family^* = \{A \subseteq S : S \setminus A \not\in \family\}\]
    \begin{itemize}
        \item Proof: \cref{thm_equiv_fam_dual_formulation}
    \end{itemize}
    
    \item $*$ is an involution on families: for all families $\family$, $(\family^*)^* = \family$
    \begin{itemize}
        \item Proof: \cref{thm_dual_is_involution_on_fams}
        \item There is a more general statement we might favor: $(\family^*)^*$ is the upward-closure of $\family$.
    \end{itemize}

    \item $*: \families{S} \to \families{S}$ is a bijection
    \begin{itemize}
        \item Proof: \cref{thm_dual_is_bijection_on_fams}
    \end{itemize}

    \item For all families $\family$, $\family$ is partition regular if and only if $\family^*$ is a filter
    \begin{itemize}
        \item Proof: \cref{thm_pr_iff_dual_is_filter}
    \end{itemize}
\end{enumerate}

\subsection{Family join}

\begin{definition}[Family join]
    The \emph{family join} map $\classcap: \mathcal{P}(\mathcal{P}(S))^2 \to \mathcal{P}(\mathcal{P}(S))$ is defined by
    \[\family \classcap \familytwo = \begin{cases} \{A \cap B : A \in \family, \ B \in \familytwo\} & \text{if $\family \neq \emptyset$ and $\familytwo \neq \emptyset$} \\ \familytwo & \text{if $\family = \emptyset$} \\ \family & \text{if $\familytwo = \emptyset$} \end{cases}\]
    The (family) join of collections $\family, \familytwo \in \collections{S}$ is $\family \classcap \familytwo$.
\end{definition}

\vskip0.5cm

Facts on the level of collections:
\begin{enumerate}
    \item $\family \cup \familytwo \subseteq \family \classcap \familytwo$
    \begin{itemize}
        \item Proof: \cref{thm_classcap_contains_union}
    \end{itemize}

    \item $\classcap$ is commutative: $\family \classcap \familytwo = \familytwo \classcap \family$
    \begin{itemize}
        \item Proof: \cref{thm_classcap_commutative}
    \end{itemize}
    
    \item $\classcap$ is associative: $\family \classcap (\familytwo \classcap \familythree) = (\family \classcap \familytwo) \classcap \familythree$
    \begin{itemize}
        \item Proof: \cref{thm_classcap_associative}
    \end{itemize}
    
    \item $\classcap$ is monotone: if $\family_1 \subseteq \family_2$ and $\familytwo_1 \subseteq \familytwo_2$, then $\family_1 \classcap \familytwo_1 \subseteq \family_2 \classcap \familytwo_2$.
    \begin{itemize}
        \item Proof: \cref{thm_classcap_monotone}

\end{enumerate}

\subsubsection{Dual facts}

\begin{theorem}
    \label{thm_dual_of_fam_is_fam}
    \lean{Family.famDual}
    If $\family$ is a family, then $\family^*$ is a family.  That is, $\cdot^*: \families{S} \to \families{S}$.
\end{theorem}

\begin{proof}
    By the definition of family, we need only check that $\family^*$ is an upward closed collection of subsets of $S$.  It is, by definition, a collection of subsets of $S$.  If $A \in \family^*$ and $A \subseteq A' \subseteq S$, to show that $A' \in \family^*$, we need only to show that for all $B \in \family$, $A' \cap B \neq \emptyset$.  Since $A \in \family^*$, we have that $\emptyset \neq A \cap B \subseteq A' \cap B$, as desired.
\end{proof}

\begin{theorem}

\subsubsection{Family join}

\begin{theorem}
\label{thm_family_classcap_is_family}
    If $\family$ and $\familytwo$ are families, then $\family \classcap \familytwo$ is a family.
\end{theorem}

\begin{proof}
    By definition, $\family \classcap \familytwo$ is a collection of subsets of $S$.  We need to show that it is upward closed.

    If $\family = \emptyset$ or $\familytwo = \emptyset$, then $\family \classcap \familytwo$ is a family, and hence is upward closed.  Assume $\family \neq \emptyset$ and $\familytwo \neq \emptyset$.  We will show that $\family \classcap \familytwo$ is upward closed.

    Let $A \in \family \classcap \familytwo$ and $A \subseteq A' \subseteq S$.  We must show that $A' \in \family \classcap \familytwo$.  By definition, there exists $B \in \family$ and $C \in \familytwo$ such that $A = B \cap C$.  Define $B' = B \cup A'$ and $C' = C \cup A'$.

    We claim that
    \begin{enumerate}
        \item $B' \in \family$,
        \item $C' \in \familytwo$,
        \item $B' \cap C' = A'$.
    \end{enumerate}
    If we show all three, then we are done, since it will follow that $A' \in \family \classcap \familytwo$.
    
    (1) Since $B \in \family$, $\family$ is upward closed, and $B \subseteq B' \subseteq S$, we have that $B' \in \family$.

    (2) Since $C \in \familytwo$, $\familytwo$ is upward closed, and $C \subseteq C' \subseteq S$, we have that $C' \in \familytwo$.

    (3) From set theory, union distributes over intersection ($X \cup (Y \cap Z) = (X \cup Y) \cap (X \cup Z)$).  We see
    \[B' \cap C' = (B \cup A') \cap (C \cup A') = (B \cap C) \cup A' = A \cup A'.\]
    Since $A \subseteq A'$, $A \cup A' = A'$, as desired.
\end{proof}

\begin{theorem}
    \label{thm_classcap_contains_union}
    $\family \cup \familytwo \subseteq \family \classcap \familytwo$
\end{theorem}

\begin{proof}
    If $\family = \emptyset$ or $\familytwo = \emptyset$, this is immediate by definition.
    
    Assume $\family \neq \emptyset$ and $\familytwo \neq \emptyset$. We will show $\family \subseteq \family \classcap \familytwo$ and $\familytwo \subseteq \family \classcap \familytwo$.

    Let $A \in \family$.  Since $\familytwo \neq \emptyset$ and $\familytwo$ is upward closed, $S \in \familytwo$.  By definition $A = A \cap S \in \family \classcap \familytwo$.  Therefore, $\family \subseteq \family \classcap \familytwo$.

    Let $B \in \familytwo$.  Since $\family \neq \emptyset$ and $\family$ is upward closed, $S \in \family$.  By definition $B = S \cap B \in \family \classcap \familytwo$. Therefore, $\familytwo \subseteq \family \classcap \familytwo$.
\end{proof}

\begin{theorem}[$\classcap$ is commutative]
    \label{thm_classcap_commutative}
    For all $\family, \familytwo \in \collections{S}$,
    \[\family \classcap \familytwo = \familytwo \classcap \family.\]
\end{theorem}

\begin{proof}
    Let $\family, \familytwo \in \collections{S}$.  We must show that
    \[\family \classcap \familytwo = \familytwo \classcap \family.\]
    
    If $\family = \emptyset$ or $\familytwo = \emptyset$, this is immediate by definition.  Assume $\family \neq \emptyset$ and $\familytwo \neq \emptyset$ so that
    \[\family \classcap \familytwo = \{A \cap B : A \in \family, \ B \in \familytwo\}.\]
    Let $X \subseteq S$. Each line is equivalent to the one that follows it.
    \begin{enumerate}
        \item $X \in \family \classcap \familytwo$
        \item there exists $A \in \family, \ B \in \familytwo$ such that $X = A \cap B$
        \item there exists $A \in \family, \ B \in \familytwo$ such that $X = B \cap A$
        \item $X \in \familytwo \classcap \family$
    \end{enumerate}
    Therefore, $\family \classcap \familytwo = \familytwo \classcap \family$, as desired.
\end{proof}

\begin{theorem}[$\classcap$ is associative]
    \label{thm_classcap_associative}
    For all $\family, \familytwo, \familythree \in \collections{S}$,
    \[(\family \classcap \familytwo) \classcap \familythree = \family \classcap (\familytwo \classcap \familythree).\]
\end{theorem}

\begin{proof}
    Let $\family, \familytwo, \familythree \in \collections{S}$.  We must show that
    \[(\family \classcap \familytwo) \classcap \familythree = \family \classcap (\familytwo \classcap \familythree).\]
    If $\family = \emptyset$ or $\familytwo = \emptyset$ or $\familythree = \emptyset$, it is quick to check that this equality holds.  Therefore, we will assume $\family, \familytwo, \familythree \neq \emptyset$.

    Since $\family, \familytwo \neq \emptyset$,
    \[\family \classcap \familytwo = \{A \cap B : A \in \family, \ B \in \familytwo\}.\]
    Similarly, since $\familytwo, \familythree \neq \emptyset$,
    \[\familytwo \classcap \familythree = \{A \cap B : A \in \familytwo, \ B \in \familythree\}.\]

    Let $X \subseteq S$. Each line is equivalent to the one that follows it.
    \begin{enumerate}
        \item $X \in (\family \classcap \familytwo) \classcap \familythree$
        \item there exists $A \in (\family \classcap \familytwo)$ and $B \in \familythree$ such that $X = A \cap B$
        \item there exists $C \in \family, \ D \in \familytwo$, and $B \in \familythree$ such that $X = (C \cap D) \cap B$
        \item there exists $C \in \family, \ D \in \familytwo$, and $B \in \familythree$ such that $X = C \cap (D \cap B)$
        \item there exists $C \in \family$ and $E \in \familytwo \classcap \familythree$ such that $X = C \cap E$
        \item $X \in \family \classcap (\familytwo \classcap \familythree)$
    \end{enumerate}
    Therefore, $(\family \classcap \familytwo) \classcap \familythree = \family \classcap (\familytwo \classcap \familythree)$, as desired.
\end{proof}

\begin{theorem}[$\classcap$ is monotone]
    \label{thm_classcap_monotone}
    For all $\family_1, \family_2, \familytwo_1, \familytwo_2 \in \collections{S}$, if $\family_1 \subseteq \family_2$ and $\familytwo_1 \subseteq \familytwo_2$, then $\family_1 \classcap \familytwo_1 \subseteq \family_2 \classcap \familytwo_2$.
\end{theorem}

\begin{proof}
\uses{thm_classcap_contains_union}
    If $\family_1 = \emptyset$, then by the definition of $\classcap$ and \cref{thm_classcap_contains_union},
    \[\family_1 \classcap \familytwo_1 = \familytwo_1 \subseteq \family_2 \cup \familytwo_2 \subseteq \family_2 \classcap \familytwo_2.\]
    By the analogous argument, we reach the same conclusion if $\familytwo_1 = \emptyset$.

    If $\family_2 = \emptyset$ or $\familytwo_2 = \emptyset$, then $\family_1 = \emptyset$ or $\familytwo_1 = \emptyset$, and we reach the desired conclusion by the previous paragraph.

    Therefore, we may assume that none of $\family_1, \family_2, \familytwo_1, \familytwo_2$ are empty.  Let $X \subseteq S$.  Each line follows from the previous one:
    \begin{enumerate}
        \item $X \in \family_1 \classcap \familytwo_1$
        \item there exists $A \in \family_1$ and $B \in \familytwo_1$ such that $X = A \cap B$
        \item there exists $A \in \family_2$ and $B \in \familytwo_2$ such that $X = A \cap B$
        \item $X \in \family_2 \classcap \familytwo_2$
    \end{enumerate}
    Therefore, $\family_1 \classcap \familytwo_1 \subseteq \family_2 \classcap \familytwo_2$, as was to be shown.
\end{proof}

\begin{leanrmk}
    The topological dynamical systems in Mathlib are called Flows.  See \url{https://leanprover-community.github.io/mathlib4_docs/Mathlib/Dynamics/Flow.html}.  A flow is an action of a topological monoid on a topological space by continuous maps.  Since we care only about discrete semigroups (monoids), we are likely better off defining our own class of topological dynamical systems.  We may still find it useful to use the notion of invariance, here: \url{https://leanprover-community.github.io/mathlib4_docs/Mathlib/Dynamics/Flow.html}
\end{leanrmk}

\begin{leanrmk}
    Question: Do we define topological dynamical systems as semigroup actions or monoid actions?  The only difference is that a monoid action requires that the identity element of the monoid acts by the identity map on the space.  This is a natural assumption, but may not be required for us.\\
    
    Answer: semigroup actions.  It seems we can do everything we need to do while avoiding monoids, and especially monoid actions.\\

    Question: Do we define topological dynamical systems with phase space ``compact Hausdorff'' or ``uniform''?  Uniform is more general, but perhaps is not necessary for our purposes.\\
    
    Answer: compact Hausdorff phase spaces\\

    Question: A ``system'' for us is usually a bundle of a phase space $X$ together with an action of a semigroup $S$.  But in our applications, it may be more efficient to un-bundle these by focusing on the action as the object.  So, instead, we would say: $T$ is an action of $S$ on $X$.  Then $T$ is minimal; a subset of $X$ is $T$-invariant; if $R$ is an action of $S$ on $Y$, then a surjection $\pi: X \to Y$ is a factor map if $S \circ \pi = \pi \circ T$; etc...\\
    
    Answer: since the data provided is essentially the same, I don't right now see a meaningful difference here.  Let's go with the more recognizable ``system'' as a bundling of space and action.
\end{leanrmk}

\begin{definition}[Semigroup action]
\lean{SemigroupAction}
\leanok{}
Let $S$ be a semigroup and $X$ be a set.  An action of $S$ on $X$ is a map $f: S \to X \to X$ such that for all $s, t \in S$,
\[(f s) \circ (f t) = f (s*t).\]
\end{definition}

\begin{definition}[Topological dynamical system]
\lean{DynamicalSystem}
\leanok{}
    Let $S$ be a non-empty semigroup and $X$ be a non-empty, compact Hausdorff space. An \emph{$S$-system} is an action of $S$ on $X$ such that for all $s \in S$, $f s$ is continuous.
\end{definition}

Notation: for $x \in X$, $U \subseteq X$, $g \in G$, and $H \subseteq G$, we write

    \item $gU$ to denote the set $\{g x \mid x \in U\}$;

    \item $GU$ to denote the set $\{gx \mid g \in G, \ x \in U\}$.
\end{enumerate}

\begin{theorem}[Orbit closure transitivity]
\label{theorem_orbit_closure_transitivity}
\lean{orbitTransitivity}
\leanok{}
    Let $X$ be an $S$-system and $x,y,z \in X$. If $y \in \overline{Sx}$ and $z \in \overline{Sy}$, then $z \in \overline{Sx}$.
\end{theorem}

\begin{leanrmk}
    Here we will use \verb|mem_closure_iff| repeatedly.
\end{leanrmk}

\begin{proof}
\leanok{}
    Let $U \ni z$.  Since $z \in \overline{Sy}$, $U \cap Sy \neq \emptyset$, so there exists $s \in S$ such that $y \in s^{-1}U$.  Since $y \in \overline{Sx}$ and $s^{-1}U$ is open, $s^{-1} U \cap Sx \neq \emptyset$.  Thus, there exists $t \in S$ such that $tx \in s^{-1} U$.  Therefore, $stx \in U$, and so $U \cap Sx \neq \emptyset$.  Since $U$ was arbitrary, we get that $z \in \overline{Sx}$, as was to be shown.
\end{proof}

\begin{lemma}
\label{lemma_closure_of_inv_is_inv}
\leanok{}

\subsection{Minimality}

\begin{definition}[Minimal system]
    % A $S$-system $X$ is \emph{minimal} if for all $x \in X$, the set $Sx$ is dense in $X$.
    An $S$-system $X$ is \emph{minimal} if the only non-empty, closed, $S$-invariant subset of $X$ is $X$.
\end{definition}

\begin{theorem}[Every system has a minimal subsystem]
\label{thm_every_system_has_minimal_subsystem}
    Let $S$ be a semigroup and $X$ be an $S$-system.  There exists a non-empty, closed, $S$-invariant subset $Z \subseteq X$ (that is, an $S$-subsystem $Z$ of $X$) such that the $S$-system $Z$ is minimal.
\end{theorem}

\begin{proof}
\uses{lemma_intersection_of_invariant}
    The subsystems of $X$ are ordered by containment.  Given any collection $Z_i$, $i \in I$, of $S$-subsystems of $X$, $\cap_i Z_i$ is closed, non-empty (by compactness), and $S$-invariant (by \cref{lemma_intersection_of_invariant}). Thus, $\cap_i Z_i$ is an $S$-subsystem of $X$.  This shows that every totally ordered subset of the set of subsystems of $X$ has a lower bound in $X$.  By Zorn's lemma, the set of subsystems has a minimal element.  This minimal element is a subsystem of $X$ that is minimal.
\end{proof}

\begin{theorem}[Minimal iff dense orbits]
\label{thm_minimal_iff_dense_orbits}
    Let $S$ be a semigroup and $X$ be an $S$-system.  The system $X$ is minimal if and only if for all $x \in X$, $\overline{Sx} = X$.
\end{theorem}

\begin{proof}
\uses{lemma_orb_closure_is_invariant}
    By \cref{lemma_orb_closure_is_invariant}, $\overline{Sx}$ is an $S$-system.  If $X$ is minimal, then $\overline{Sx} = X$, so $x$ has a dense orbit.

    On the other hand, if $Z$ is a closed, non-empty, $S$-invariant set, then choosing $x \in Z$, we see by the dense-orbit assumption that
    \[X = \overline{Sx} \subseteq \overline{SZ} \subseteq Z \subseteq X.\]
    It follows that $Z = X$, and hence that $X$ is a minimal $S$-system.
\end{proof}

\begin{lemma}[Images of minimal systems are minimal]
\label{lemma_img_of_min_is_min}
    Let $S$ be a semigroup and $\pi: X \to Y$ be a factor map of $S$-systems. If $X$ is minimal, then $Y$ is minimal.
\end{lemma}

\begin{proof}
    Let $Z \subseteq Y$ be a non-empty, closed, $S$-invariant subset of $Y$.  Then $\pi^{-1} Z$ is a non-empty, closed, $S$-invariant subset of $X$.  By the minimality of $X$, $\pi^{-1} Z = X$.  Since $\pi$ is surjective, $Z = Y$, as desired.
\end{proof}

\begin{lemma}[Minimal actions act surjectively]
\label{lemma_minimal_comm_action_implies_surjective}
    Let $S$ be a commutative semigroup.  Let $X$ be a minimal $S$-system.  For all $s \in S$, the map $s: X \to X$ is surjective.
\end{lemma}

\begin{proof}
    Let $s \in S$.  Since $S$ is commutative, the set $sX$ is a closed, $S$-invariant subset of $X$.  By minimality, $sX = X$, showing that $s$ is surjective, as desired.
\end{proof}

\subsection{Uniform recurrence}

\begin{definition}[Uniformly recurrent point]
    Let $S$ be a semigroup, $X$ be an $S$-system, and $x \in X$.  The point $x$ is $S$-uniformly recurrent if for all non-empty, open $U \ni x$, the set $R(x,U)$ is syndetic.
\end{definition}

\begin{lemma}
    \label{lemma_pts_in_minimal_are_ur}
    Let $S$ be a semigroup and $X$ be a minimal $S$-system. For all $x \in X$, $x$ is $S$-uniformly recurrent.
\end{lemma}

\begin{proof}
\uses{thm_minimal_iff_dense_orbits,lemma_return_to_union,lemma_return_time_algebra_one}
    Let $x \in U \subseteq X$. By \cref{thm_minimal_iff_dense_orbits}, all orbits are dense, so for $y \in X$, there exists $s_y \in S$ such that $sy \in U$.  Therefore, $X = \cup_{y \in X} s_y^{-1}U$.  By compactness, there exists a finite set $F \subseteq X$ such that $X = \cup_{y \in F} s_y^{-1} U$.  By \cref{lemma_return_to_union} and \cref{lemma_return_time_algebra_one},
    \[S = R(x,X) = R(x, \bigcup_{y \in F} s_y^{-1} U ) = \bigcup_{y \in F} R(x, s_y^{-1} U) = \bigcup_{y \in F} s_y^{-1} R(x, U).\]
    By definition, the set $R(x,U)$ is syndetic.
\end{proof}

\begin{lemma}
    \label{lemma_orb_closure_of_ur_is_minimal}
    Let $S$ be a semigroup and $X$ be an $S$-system.  If $x \in X$ is $S$-uniformly recurrent, then $\overline{Sx}$ is a minimal $S$-system containing $x$.
\end{lemma}

\begin{proof}
\uses{lemma_orb_closure_is_invariant}
    Note that $\overline{Sx}$ is an $S$-system by \cref{lemma_orb_closure_is_invariant}. For all $U \ni x$, $R(x,U)$ is syndetic, hence non-empty.  Therefore, $x \in \overline{Sx}$.
    
    Let $y \in \overline{Sx}$ and $U \subseteq \overline{Sx}$ be open and non-empty.  We must show that there exists $s \in S$ such that $sy \in U$.

    Let $V \subseteq U$ be non-empty, open, and $\gamma \subseteq X^2$ be an entourage such that if $(a,b) \in \gamma$ and $a \in V$, then $b \in U$.
    
    Since $V \subseteq \overline{Sx}$ is open, non-empty, there exists $t \in S$ such that $tx \in V$, that is, $x \in t^{-1} V$.  By assumption $R(x,t^{-1} V)$ is syndetic: there exists a finite set $F \subseteq S$ such that for all $s \in S$, there exists $f \in F$ such that $fs \in R(x,t^{-1}U)$.

    By the uniform continuity of the maps $tf: X \to X$, $f \in F$, there exists an entourage $\alpha \subseteq X^2$ such that for all $f \in F$, $tf \alpha \subseteq \gamma$.  Since $y \in \overline{Sx}$, there exists $s \in S$ such that $(sx,y) \in \alpha$.  By syndeticity, there exists $f \in F$ such that $fs \in R(x,t^{-1}V)$, that is, $tfs x \in V$.

    Since $(sx,y) \in \alpha$, we have that $(tfsx,tfy) = tf(sx,y) \in tf \alpha \subseteq \gamma$.  Since $tfsx \in V$, by our choice of $V$ and $\gamma$, we have that $tfy \in U$.  Therefore, $tf \in R(y,U)$, and hence $R(y,U)$ is non-empty, as was to be shown.
\end{proof}

\begin{lemma}
    \label{lemma_in_orb_clos_of_ur_implies_ur}
    Let $S$ be a semigroup, $X$ be an $S$-system, and $x, y \in X$. If $x$ is $S$-uniformly recurrent and $y \in \overline{Sx}$, then $y$ is $S$-uniformly recurrent. 
\end{lemma}

\begin{proof}
\uses{lemma_orb_closure_of_ur_is_minimal,lemma_pts_in_minimal_are_ur}
    By \cref{lemma_orb_closure_of_ur_is_minimal}, $\overline{Sx}$ is a minimal $S$-system.  By \cref{lemma_pts_in_minimal_are_ur}, all points of $\overline{Sx}$ are $S$-uniformly recurrent. \dgg{Something subtle here that may cause issues in Lean: $y$ being $S$-uniformly recurrent is independent of the system. \end{proof}

\begin{lemma}
    \label{lemma_dense_S_minimal_points_in_X2}
    Let $S$ be a commutative semigroup, and let $X$ be a minimal $S$-system.  The set of points $(x,y) \in X^2$ that are $S$-uniformly recurrent is dense in $X^2$.
\end{lemma}

\begin{proof}
\uses{thm_every_system_has_minimal_subsystem,lemma_img_of_inv_is_inv,lemma_img_of_min_is_min,lemma_pts_in_minimal_are_ur}
    Let $U \times V \subseteq X^2$ be a non-empty, open set. We will show that $U \times V$ contains an $S$-uniformly recurrent point.
    
    By \cref{thm_every_system_has_minimal_subsystem}, there exists a closed, non-empty, $S$-invariant (under the diagonal action) set $Z \subseteq X^2$.  Note that $\pi_1: X^2 \to X$ is an $S$-system factor map.  By \cref{lemma_img_of_inv_is_inv}, $\pi_1 Z$ is a closed, non-empty, $S$-invariant subset of $X$.  Since $X$ is minimal, $\pi_1 Z = X$.  Therefore, there exists $(x,y) \in Z$ such that $x \in U$.
    
    Since $X$ is a minimal $S$-system, there exists $s \in S$ such that $sy \in V$.  Note that the point $(x,sy) \in U \times V$.  We have only left to show that $(x,sy)$ is $S$-uniformly recurrent.
    
    Since $S$ is commutative, the map $X^2 \to X^2$ given by $(w,z) \mapsto (w,sz)$ commutes with the diagonal action of $S$.  By \cref{lemma_img_of_inv_is_inv}, the image of $Z$ under this map is an $S$-system.  By \cref{lemma_img_of_min_is_min}, the image of $Z$ under this map is a minimal $S$ system.  By \cref{lemma_pts_in_minimal_are_ur}, the point $(x,sy)$ is $S$-uniformly recurrent, as desired.
\end{proof}


\subsection{Proximality}

\begin{leanrmk}

\end{leanrmk}

\begin{definition}[{Proximal, cf. \cite[Ch. 4, (2.11)]{devries_book_1993}}]
\label{def_proximal}
    Let $X$ be an $S$-system and $x, y \in X$.  The point $x$ is \emph{proximal to $y$} if for all $\alpha$ in the uniform structure on $X$, there exists $s \in S$ such that $(sx,sy) \in \alpha$.
\end{definition}

\begin{theorem}
\label{theorem_proximal_and_minimal_implies_orbit_closure_contains_diagonal}
    Let $S$ be a semigroup, $X$ be a minimal $S$-system, and $x, y \in X$.  If $x$ and $y$ are $S$-proximal, then $\Delta(X)$ is contained in the $S$-orbit-closure of $(x,y)$.
\end{theorem}

\begin{proof}
\uses{theorem_orbit_closure_transitivity}
    By the definition of proximal, for all entourages $\alpha \subseteq X^2$, there exists $s \in S$ such that $(sx,sy) \in \alpha$.  The net $\alpha \mapsto (s_\alpha x, s_\alpha y)$ has a cluster point.  Since the cluster point is in every entourage, it belongs to $\Delta(X)$.  Therefore, there exists $z \in X$ such that $(z,z)$ is in the $S$-orbit closure of $(x,y)$.  By minimality, every point $x$ is in the $S$-orbit closure of $z$, so $\Delta(X)$ is in the $S$-orbit closure of $(z,z)$.  By orbit closure transitivity (\cref{theorem_orbit_closure_transitivity}), we have that $\Delta(X)$ is contained in the $S$-orbit closure of $(x,y)$, as desired.
\end{proof}

\subsection{Regional proximality}

\begin{definition}[{Regionally proximal relation, cf. \cite[Ch. 5, Section 1]{devries_book_1993}}]
\label{def_rp_relation}
    Let $S$ be a semigroup and $X$ be an $S$-system.  The \emph{regionally proximal relation} is defined as
    \[\RP = \bigcap_{\alpha} \overline{S \alpha},\]
    where the intersection is over all $\alpha$ in the uniformity on $X$, and $S$ acts diagonally.  We say that point $x \in X$ is \emph{regionally proximal to} point $y \in X$ if $(x,y) \in \RP$.
\end{definition}

\begin{lemma}
\label{lemma_rp_basics}
    Let $X$ be an $S$-system.  The relation $\RP$ is reflexive, symmetric, invariant, and closed.
\end{lemma}

\begin{proof}
\uses{lemma_intersection_of_invariant}
    \begin{itemize}
        \item Symmetric: Denote by $\Phi: X^2 \to X^2$ the map $\Phi(x,y) = (y,x)$. Let $(x,y) \in \RP$ and $\alpha \subseteq X^2$ be an entourage. By the definition of a uniform space, $\Phi \alpha$ is an entourage (sometimes called $\alpha^{-1}$). Since $(x,y) \in \RP$, $(x,y) \in \overline{G \Phi \alpha}$. But $\overline{G \Phi \alpha} = \Phi \overline{G \alpha}$. Therefore, $(y,x) \in \overline{G \alpha}$, as was to be shown.

        \item Invariant: The set $G\alpha$ is clearly $G$-invariant.  Since $G$ acts by continuous maps,
        \[g \overline{G\alpha} \subseteq \overline{gG \alpha} \subseteq \overline{G \alpha},\]
        whereby $\overline{G\alpha}$ is $G$-invariant.  Finally, by \cref{lemma_intersection_of_invariant}, an arbitrary intersection of $G$-invariant sets is $G$-invariant, so $\RP$ is $G$-invariant.

        \item Closed: an arbitrary intersection of closed sets is closed.
    \end{itemize}   
\end{proof}

\begin{lemma}
\label{lemma_rp_reflexive}
    Let $X$ be an $S$-system.  If $\overline{SX} = X$, then $\RP$ is reflexive.
\end{lemma}

\begin{proof}
    Let $\alpha \subseteq X^2$ be an entourage.  Since $\overline{SX} = X$, $\Delta \subseteq \overline{S \alpha}$. Since $\alpha$ was arbitrary, $\Delta \subseteq \RP$.
\end{proof}

\begin{definition}[Diagonal limit point set]
\label{def_diag_limit_pt_set}

\end{definition}

\begin{lemma}
\label{lemma_diag_limit_pt_set}
    Let $S$ be a semigroup, $X$ be an $S$-system, $(x,y) \in X^2$, and $A, B \subseteq S$.  If $A \subseteq B$, then $\Omega_A(x,y) \subseteq \Omega_B(x,y)$.
\end{lemma}

\begin{lemma}
\label{lemma_rp_implies_net_iff}
    Let $S$ be a semigroup, $X$ be an $S$-system, and $x, y \in X$. For all $A \subseteq S$,
    \[\bigcap_{\alpha} \overline{A \alpha} = \{(x,y) \in X^2 : \Omega_A(x,y) \neq \emptyset\}.\]
    In particular, setting $A = S$, we see that $(x,y) \in \RP$ if and only if there exists $w \in X$ and a net $(a_n, b_n, g_n)_n \subseteq X \times X \times S$ such that
    \begin{align*}
        \lim_{n \to \infty} (a_n, b_n, g_n a_n, g_n b_n) &= (w, w, x, y).
    \end{align*}
\end{lemma}

\begin{proof}
    Let $A \subseteq S$.  Suppose $(x,y) \in \cap_{\alpha} \overline{A \alpha}$. Let $D$ be the directed set consisting of Cartesian products of open neighborhoods of $x$, open neighborhoods of $y$, and entourages of $X$. For $n = (U,V,\alpha) \in D$, we see from
    \[(x,y) \in \bigcap_{\alpha} \overline{A \alpha}\]
    that
    \[(U \times V) \cap A \alpha \neq \emptyset.\]
    Thus, there exists $(a_n,b_n) \in \alpha$ and $s_n \in A$ such that $(s_n w_n, s_n x_n) \in U \times V$.  By compactness, the net $n \mapsto (a_n, b_n, s_n a_n, s_n b_n)$ has a cluster point.  Thus, there exists $w \in X$ such that $(a_n,b_n)$ converges to $(w,w)$ and $(s_n a_n, s_n b_n)$ converges to $(x,y)$.

    Conversely, suppose that $(x,y) \in X^2$ is such that $\Omega_A(x,y) \neq \emptyset$: there exists $w \in X$ and a net $(a_n, b_n, s_n)_n \subseteq X \times X \times A$ such that
    \begin{align*}
        \lim_{n \to \infty} (a_n, b_n, s_n a_n, s_n b_n) &= (w, w, x, y).
    \end{align*}
    We will show that $(x,y) \in \cap_{\alpha} \overline{A \alpha}$. Let $\alpha \subseteq X^2$ be an entourage and $W \ni (x,y)$ be open.  Then $\alpha \times W$ is an open neighborhood of $(w,w,x,y)$, so there exists $n$ such that $(a_n,b_n,s_na_n,s_n,b_n) \in \alpha \times W$.  In particular, $W \cap A \alpha \neq \emptyset$.  Since $W$ and $\alpha$ were arbitrary, we see that $(x,y) \in \overline{A \alpha}$, as was to be shown.
\end{proof}

    Suppose $w \in \Omega_A(x,y)$.  Clearly $(w,w) \in \Delta(X)$, so we need only show that $(w,w) \in \cap_{U} \overline{A^{-1}U}$.  Let $U \subseteq X^2$ be open and contain $(x,y)$. There is a net $(a_n, b_n, s_n) \in X^2 \times A$ such that
    \[\lim_{n \to \infty} (a_n, b_n, s_n a_n, s_n b_n) = (w, w, x, y).\]
    Let $W \ni (w,w)$ be an open subset of $X^2$.  Since $W \times U$ is an open neighborhood of $(w, w, x, y)$, there is $n$ such that $(a_n,b_n) \in W$ and $s_n(a_n,b_n) \in U$.  Therefore,
    \[(a_n, b_n) \in s_n^{-1} U \cap W,\]
    and so $A^{-1} U \cap W \neq \emptyset$.  Since $W \ni (w,w)$ was arbitrary, we see that $(w,w) \in \overline{A^{-1}U}$, as was to be shown.

    Conversely, suppose $(w,w)$ is in the right hand side. Let $D$ be the product of the set of open neighborhoods of $(w,w)$ and the set of open neighborhoods of $(x,y)$. For $d = (W,U)$, we get $A^{-1} U \cap W \neq \emptyset$.  Therefore, there exists $(a_d,b_d) \in W$ and $s_d \in A$ such that $s_d(a_d,b_d) \in U$.  The net $(a_d,b_d,s_da_d, s_d,b_d)$ converges to $(w,w,x,y)$, showing that $w \in \Omega_A(x,y)$, as desired.
\end{proof}

\begin{lemma}
    \label{lemma_S_ideal_description_of_rp}
    Let $S$ be a commutative semigroup, $X$ be a minimal $S$-system, and $x, y \in X$. For all $t \in S$,
    \[\RP = \bigcap_{\alpha} \overline{St \alpha}.\]
\end{lemma}

\begin{proof}
\uses{lemma_dense_S_minimal_points_in_X2,lemma_orb_closure_of_ur_is_minimal,lemma_minimal_comm_action_implies_surjective}
    Let $t \in S$.  Since $St \subseteq S$, we have by the definition of $\RP$ that
    \[\bigcap_{\alpha} \overline{S t \alpha} \subseteq \RP.\]
    Conversely, suppose $(x,y) \in \RP$.  Let $\alpha \subseteq X^2$ be an entourage, and let $W \ni (x,y)$ be open.  We must show that $W \cap St \alpha \neq \emptyset$.

    Since $(x,y) \in \RP$, we have that $W \cap S \alpha \neq \emptyset$.  Thus, there exists $s \in S$ such that $W \cap s \alpha \neq \emptyset$.  Therefore, $s^{-1} W \cap \alpha \neq \emptyset$.  Since this is an open set, by \cref{lemma_dense_S_minimal_points_in_X2}, there exists $(a,b) \in s^{-1} W \cap \alpha$ that is $S$-uniformly recurrent (under the diagonal action).  It follows that
    \[(a,b) \in \overline{S(a,b)}.\]
    Since $(a,b)$ is $S$-uniformly recurrent, it follows by \cref{lemma_orb_closure_of_ur_is_minimal} that $\overline{S(a,b)}$ is a minimal $S$-system.  By \cref{lemma_minimal_comm_action_implies_surjective} and the fact that $S$ is commutative, we have that
    \[\overline{St(a,b)} = \overline{tS(a,b)} = t\overline{S(a,b)} = \overline{S(a,b)}.\]
    Therefore, $(a,b) \in s^{-1}W \cap \alpha \cap \overline{St(a,b)}$ and so this set is non-empty.  Since $s^{-1}W \cap \alpha$ is open, $s^{-1}W \cap \alpha \cap St(a,b) \neq \emptyset$.  Therefore, there exists $\ell \in S$ such that $\ell t (a,b) \in s^{-1}W \cap \alpha$.  That is: $s \ell t (a,b) \in W$, whereby $S t (a,b) \cap W \neq \emptyset$.  Since $(a,b) \in \alpha$, we have shown that $S t \alpha \cap W \neq \emptyset$, as was to be shown.
\end{proof}

\begin{lemma}
\label{lemma_rp_implies_net_minimal}
    Let $S$ be a commutative semigroup, $X$ be a minimal $S$-system, and $(x, y) \in X^2$. If $(x,y) \in \RP$, then $\Omega_S(x,y) = X$, that is, for all $w \in X$, there exists a net $(a_n, b_n, s_n)_n \subseteq X \times X \times S$ such that
    \begin{align*}
        \lim_{n \to \infty} (a_n, b_n, s_n a_n, s_n b_n) &= (w, w, x, y).
    \end{align*}
\end{lemma}

\begin{proof}
\uses{lemma_S_ideal_description_of_rp,lemma_diag_limit_pt_set,lemma_set_of_ws_is_closed}
    Let $(x,y) \in \RP$.  First we claim that for all $t \in S$, $\Omega_{St}(x,y) \neq \emptyset$.  Indeed, by \cref{lemma_S_ideal_description_of_rp},
    \[(x,y) \in \RP = \bigcap_{\alpha} \overline{St \alpha}.\]
    By \cref{lemma_diag_limit_pt_set}, we have that $\Omega_{St}(x,y) \neq \emptyset$.

    Next we claim that for all $t \in S$, $t \Omega_{St}(x,y) \subseteq S$. Let $t \in S$ and $w \in \Omega_{St}(x,y)$.  Thus, there exists a net $(a_n, b_n, s_n t)_n \subseteq X^2 \times S$ such that
    \begin{align*}
        \lim_{n \to \infty} (a_n, b_n, s_n t a_n, s_n t b_n) &= (w, w, x, y).
    \end{align*}
    It follows that $t(a_n,b_n) \to t(w,w)$. So, the net $(ta_n, tb_n, s_n)_n \subseteq X^2 \times S$ is such that
    \[\lim_{n \to \infty} (t a_n, t b_n, s_n t a_n, s_n t b_n) = (t w, t w, x, y).\]
    Therefore, $tw \in \Omega_S(x,y)$, as desired.

    Now we will show that $\Omega_S$ is dense in $X$. Let $U \subseteq X$ be open and non-empty.  By minimality, there exists $F \subseteq S$ finite such that $F^{-1} U = X$.  Let $t$ be the product of the elements of $F$.  By the paragraph above, there exists $w \in \Omega_{St}$.  By minimality, there exists $f \in F$ such that $fw \in U$.  Since $St \subseteq Sf$, $w \in \Omega_{St} \subseteq \Omega_{Sf}$ (by \cref{lemma_diag_limit_pt_set}).  By the paragraph above, $f \Omega_{Sf} \subseteq \Omega_S$, so $fw \in \Omega_S$.  Therefore, $\Omega_S \cap U \neq \emptyset$, as was to be shown.
    
    Finally, by the description of $\Omega_S(x,y)$ in \cref{lemma_set_of_ws_is_closed}, we see that $\Omega_S$ is closed.  Since it is a closed and dense subset of $X$, we get that $\Omega_S = X$, as desired.
\end{proof}

\begin{lemma}
\label{lemma_rp_implies_intersects_open_nbhd_diagonal}
    Let $S$ be a commutative semigroup, $X$ be a minimal $S$-system, and $x, y \in X$. If $(x,y) \in \RP$, then for all non-empty, open $U \subseteq X$, and all non-empty, open $V \subseteq X^2$ containing $(x,y)$,
    \[S^{-1} V \cap (U \times U) \neq \emptyset.\]
\end{lemma}

\begin{proof}
\uses{lemma_rp_implies_net_minimal}
    Let $(x,y) \in \RP$.  Let $U \subseteq X$ be non-empty, open, and let $V \subseteq X^2$ be open and contain $(x,y)$.  Let $w \in U$.  It follows from \cref{lemma_rp_implies_net_minimal} that there exists a net $(a_n, b_n, s_n)_n \subseteq X \times X \times S$ such that
    \begin{align*}
        \lim_{n \to \infty} (a_n, b_n, s_n a_n, s_n b_n) &= (w, w, x, y).
    \end{align*}
    Since $U \times U \times V$ is an open neighborhood of $(w,w,x,y)$, there exists $n$ such that $(a_n,b_n) \in U \times U$ and $s_n (a_n, b_n) \in V$.  Therefore,
    \[(a_n, b_n) \in s^{-1} V \cap (U \times U),\]
    and hence this set is non-empty, as was to be shown.
\end{proof}

\begin{lemma}[{cf. \cite[V. 2.12 and 2.14, pg. 416]{devries_book_1993}}]
    \label{lemma_rp_functorial_property}
    Let $X$ be a $G$-system and $\pi: X \to Y$ be a factor.  Then $(\pi \times \pi) \RP_X \subseteq \RP_Y$.
\end{lemma}

\begin{proof}
    Let $\beta \subseteq Y^2$ be an entourage.  Since $\pi: X \to Y$ is continuous and $X$ is compact, it is uniformly continuous.  In particular, there exists $\alpha \subseteq X^2$ an entourage such that $(\pi \otimes \pi) \alpha \subseteq \beta$.  We see then that
    \[(\pi \times \pi) \RP_X \subseteq (\pi \otimes \pi) \overline{G \alpha} = \overline{(\pi \otimes \pi) G \alpha} = \overline{G(\pi \otimes \pi) \alpha} \subseteq \overline{G \beta},\]
    as was to be shown.
\end{proof}



The following lemma is shown in \cite[V. 2.12 and 2.14, pg. 416]{devries_book_1993} for group actions.  Our semigroup version is considerably more involved since our semigroup maps are not necessarily open.

\begin{lemma}
    \label{lemma_rp_functorial_property_minimal}
    Let $S$ be a commutative semigroup, let $X$ be a minimal $S$-system, and let $\pi: X \to Y$ be a factor map of $S$-systems.  Then $\RP_Y = (\pi \times \pi) \RP_X$.
\end{lemma}

\begin{proof}
\uses{lemma_rp_functorial_property,lemma_image_of_entourage_somewhere_dense_along_diagonal,lemma_rp_implies_intersects_open_nbhd_diagonal}
    By \cref{lemma_rp_functorial_property}, $(\pi \times \pi) \RP_X \subseteq \RP_Y$, so we only need to prove the reverse inclusion.

    To see that $\RP_Y \subseteq (\pi \times \pi) \RP_X$, let $(x,y) \in \RP_Y$.  Let $D$ be the product of the set of entourages of $X^2$ and the set of open subsets of $Y^2$ containing $(x,y)$.  For $d = (\alpha, U)$, by \cref{lemma_image_of_entourage_somewhere_dense_along_diagonal}, there exists a non-empty, open set $V \subseteq Y$ such that $V \times V \subseteq (\pi \otimes \pi) \alpha$.  By \cref{lemma_rp_implies_intersects_open_nbhd_diagonal},
    \[S^{-1} U \cap (V \times V) \neq \emptyset.\]
    Therefore,
    \[S^{-1} U \cap (\pi \otimes \pi) \alpha \neq \emptyset.\]
    Therefore, there exists $(a_d,b_d) \in \alpha$ and $s_d \in S$ such that $s_d(\pi a_d, \pi b_d) \in U$.
    %Note that there exists $a \in X$ such that
    %\[\lim_{d} (a_d,b_d,s_d \pi a_d, s_d \pi b_d) = (a,a,x,y).\]

    By compactness of $X^4$, the net $d \mapsto (a_d,b_d,s_d a_d, s_d b_d)$ has a convergent subnet $n \mapsto (a_n,b_n,s_n a_n, s_n b_n)$.  Thus, there exists $(a,b,w,z) \in X^4$ such that
    \[\lim_n (a_n,b_n,s_n a_n, s_n b_n) = (a,b,w,z).\]
    By the construction of the net, we see that $a=b$.  Therefore, $(w,z) \in \RP_X$.
    
    Also by the construction of the net, we see that $s_n (\pi a_n, \pi b_n) \to (x,y)$.  But $s_n (\pi a_n, \pi b_n) = (\pi \otimes \pi) (s_n a_n, s_n b_n)$, and so by  continuity of $\pi \otimes \pi$, we have that
    \[(x,y) \leftarrow s_n (\pi a_n, \pi b_n) = (\pi \otimes \pi) (s_n a_n, s_n b_n) \to (\pi \otimes \pi) (w,z).\]
    Therefore, $(\pi \otimes \pi) (w,z) = (x,y)$, and so $(x,y) \in (\pi \otimes \pi) \RP_X$, as was to be shown.
\end{proof}

\subsection{Equicontinuity and regional proximality}

\begin{definition}[Equicontinuous family]
     Let $X$ and $Y$ be uniform spaces.  A set $S$ of continuous maps from $X$ to $Y$ is \emph{uniformly equicontinuous} if for every $\beta \subseteq Y^2$ in the uniform structure of $Y$ there is an $\alpha \subseteq X^2$ in the uniform structure of $X$ such that for all $f \in S$, $(f \otimes f)\alpha \subseteq \beta$.
\end{definition}

Note for the definition of equicontinuous system: A $G$-system $X$ for us is a compact Hausdorff space.  Compact Hausdorff spaces are uniform spaces with a unique uniformity given by the filter of all neighborhoods of the diagonal.

\begin{definition}[{Equicontinuous system, cf. 
\cite[Ch. 4, (2.1)]{devries_book_1993}}]
    A $G$-system $X$ is \emph{(uniformly) equicontinuous} if the set of maps $\{g: X \to X \mid g \in G\}$ is a (uniformly) equicontinuous family.
\end{definition}

\begin{lemma}
    \label{lemma_product_of_equi_is_equi}
    Let $S$ be a semigroup.  The product of equicontinuous $S$-systems is an equicontinuous $S$-system.
\end{lemma}

\begin{proof}
    If $X$ and $Y$ are $S$-systems, then $X \times Y$ is an $S$-system via the diagonal action.  We need to show that the family of maps $\{(x,y) \mapsto (gx,gy) : g \in G\}$ from $X \times Y$ to $X \times Y$ is equicontinuous.  Let $\beta \subseteq (X \times Y)^2$ be an entourage.  By some general fact concerning the uniform structure on a Cartesian product of uniform spaces (in mathlib, see \verb|entourageProd_subset|), there exist entourages $\beta_X \subseteq X^2$ and $\beta_Y \subseteq Y^2$ such that $\beta_X \times \beta_Y \subseteq \beta$.  By equicontinuity of the systems $X$ and $Y$, there exist entourages $\alpha_X \subseteq X^2$ and $\alpha_Y \subseteq Y^2$ such that for all $g \in G$, $(g \otimes g) \alpha_X \subseteq \beta_X$ and $(g \otimes g) \alpha_Y \subseteq \beta_Y$.  Then $
    \alpha \defeq \alpha_X \times \alpha_Y$ is an entourage of $X \times Y$, and for all $g \in G$, $(g \otimes g \otimes g \otimes g)\alpha \subseteq \beta$, as desired.
\end{proof}

\begin{lemma}
    \label{lemma_subsys_of_equi_is_equi}
    Let $S$ be a semigroup, and let $X$ be an equicontinuous $S$-system.  All subsystems of $X$ are equicontinuous $S$-systems.
\end{lemma}

\begin{proof}
    By assumption, the family of maps indexed by $S$ is equicontinuous.  Let $Z$ be a subsystem of $X$.  By definition, the system $Z$ is equicontinuous if the family of maps $\{\restr{s}{Z}: Z \to Z | s \in S\}$ is equicontinuous.  \dgg{This should be a general topological fact.}
\end{proof}

\begin{definition}[{Equicontinuous ICER, cf. \cite{ellis_ellis_book_2014}}]

\begin{lemma}[{cf. \cite[pg. 283]{devries_book_1993}}]
    \label{lemma_rp_and_equicontinuity}
    Let $S$ be a semigroup. An $S$-system $X$ is equicontinuous if and only if $\RP \subseteq \Delta$.

\begin{proof}

\subsection{The equicontinuous structure relation}

\begin{definition}[{cf. \cite[Ch. V, (1.3)]{devries_book_1993}}]
    Let $S$ be a semigroup and $X$ be an $S$-system.  The \emph{equicontinuous structure relation} on $X$, denoted $\seq(X)$ (or just $\seq$ if $X$ is understood), is the intersection of all equicontinuous ICERS on $X$.  (Note that $X^2$ is always one such ICER, so the intersection is non-empty.)
\end{definition}

\begin{definition}
    Let $S$ be a semigroup and $X$ be an $S$-system.  The Kronecker factor of $X$ is the quotient system $X/\seq$ together with the quotient map $\pi: X \to X/\seq$.
\end{definition}

\begin{theorem}
    Let $S$ be a semigroup and $X$ be an $S$-system.  The equicontinuous structure relation is an equicontinuous ICER.  In particular, the Kronecker factor is an equicontinuous system.
\end{theorem}

\begin{proof}
\uses{lemma_intersection_of_icers,lemma_rp_characterization_of_equicont_icer}
    The equicontinuous structure relation $\seq$ is an ICER by \cref{lemma_intersection_of_icers}.

    To show that it is an equicontinuous ICER, it suffices by \cref{lemma_rp_characterization_of_equicont_icer} to show that $\RP \subseteq \seq$.
    
    Let $R$ be an equicontinuous ICER.  We must show that $\RP \subseteq R$.  This follows by \cref{lemma_rp_characterization_of_equicont_icer}.
\end{proof}

\begin{theorem}
    Let $S$ be a semigroup and $X$ be an $S$-system.  Let $\pi: X \to K$ be the Kronecker factor.  For all factors $\sigma: X \to Z$ of $S$-systems with $Z$ equicontinuous, there exists a continuous surjection $\xi: K \to Z$ such that $\sigma = \xi \circ \pi$.  In particular, the Kronecker factor is the largest equicontinuous factor of $X$.
\end{theorem}

\begin{proof}
\uses{lemma_factors_and_icers}
    Since $\sigma: X \to Z$ is equicontinuous, $R_\sigma$ is an equicontinuous ICER.  By the definition of $\seq$, $\seq \subseteq R_\sigma$.  It follows then by \cref{lemma_factors_and_icers} that there exists a continuous surjection $\xi: K \to Z$ such that $\sigma = \xi \circ \pi$.
\end{proof}

\begin{lemma}[Principal ideals are closed]
\label{lemma_principal_ideal_is_compact}
    Let $S$ be a compact, Hausdorff right-topological semigroup, and let $t \in S$.  The set $St$ is closed.
\end{lemma}

\begin{proof}
    Since $S$ is compact and $\rho_t$ is continuous, $\rho_t(S) = St$ is compact.  Since $S$ is Hausdorff, $St$ is closed.
\end{proof}

\begin{lemma}[Left ideal contains a closed left ideal]
\label{lemma_left_ideal_contains_closed_left_ideal}
    Let $S$ be a compact, Hausdorff right-topological semigroup.  Every left ideal contains a closed left ideal.
\end{lemma}

\begin{proof}
\uses{lemma_principal_ideal_is_compact}
    Let $L$ be a left ideal.  Let $t \in L$.  Since $L$ is  a left ideal, $St$ is a left ideal contained in $L$.  By \cref{lemma_principal_ideal_is_compact}, $St$ is closed.
\end{proof}

\begin{lemma}[Minimal left ideals are compact]
\label{lemma_minimal_left_ideal_compact}
    Let $S$ be a compact, right-topological semigroup.  Minimal left ideals of $S$ are compact.
\end{lemma}

\begin{proof}
\uses{lemma_principal_ideal_is_compact}
    Let $L$ be a minimal left ideal of $S$.  Let $t \in L$. Then $St$ is a left ideal of $S$ contained in $L$.  Since $L$ is minimal, $St = L$.  By \cref{lemma_principal_ideal_is_compact}, $St$ is compact.
\end{proof}


\begin{lemma}
\label{lemma_exists_minimal_left_ideal}
    Let $S$ be a compact, Hausdorff right-topological semigroup.  Every left ideal of $S$ contains a minimal left ideal.
\end{lemma}

\begin{proof}
\uses{lemma_intersection_of_ideals_is_ideal}
    Let $L$ be a left ideal of $S$.  Consider the collection $\mathcal{C}$ of compact left ideals of $S$ contained in $L$.  For $s \in L$, we see that $Ss \subseteq L$ is a compact left ideal of $S$ contained in $L$, so $\mathcal{C}$ is non-empty.  Any totally ordered chain in $\mathcal{C}$ has a lower bound in $\mathcal{C}$, namely the intersection, which is non-empty and compact since $S$ is Hausdorff, and which is an ideal by \cref{lemma_intersection_of_ideals_is_ideal}.  By Zorn's lemma, $\mathcal{C}$ has a minimal element.  This is a minimal left ideal of $S$ contained in $L$.
\end{proof}

\begin{theorem}
    \label{exists_idempotent_in_compact_subsemigroup}
    Let $S$ be a compact, Hausdorff right-topological semigroup.  Every compact subsemigroup of $S$ contains an idempotent.
\end{theorem}

\begin{proof}
    This is precisely  \verb|exists_idempotent_in_compact_subsemigroup| in mathlib.
\end{proof}

\begin{lemma}
\label{lemma_left_ideal_contains_idempotent}
    Let $S$ be a compact, Hausdorff right-topological semigroup.  Every left ideal of $S$ contains an idempotent element.
\end{lemma}

\begin{proof}
\uses{lemma_exists_minimal_left_ideal,lemma_minimal_left_ideal_compact,exists_idempotent_in_compact_subsemigroup}
    Every left ideal contains a minimal left ideal (\cref{lemma_exists_minimal_left_ideal}).  Minimal left ideals are compact (\cref{lemma_minimal_left_ideal_compact}) subsemigroups.  Apply \cref{exists_idempotent_in_compact_subsemigroup}.
\end{proof}

\begin{definition}[Semigroup structure on $\beta S$]
\label{def_ultrafilter_multiplication}

\begin{definition}[Left/right multiplication maps]

\end{definition}

\begin{lemma}
\label{lemma_ultrafilter_mult_with_one_principal}
    Let $S$ be a semigroup, $s \in S$, and $B \subseteq S$. Then $B \in \delta_s p$ if and only if $s^{-1} B \in p$.
\end{lemma}

\begin{proof}
    Each statement is equivalent with the one following it.
    \begin{enumerate}
        \item $B \in \delta_s p$
        \item $\{a \in S : \{t \in S : at \in B \} \in p \} \in \delta_s$
        \item $\{t \in S : st \in B \} \in p$
        \item $\{t \in S : t \in s^{-1}B \} \in p$
        \item $s^{-1}B \in p$
    \end{enumerate}
\end{proof}

\begin{lemma}
\label{lemma_pull_out_inverse_mult_in_basis_set}
    Let $S$ be a semigroup, $s \in S$, and $B \subseteq S$. Then $\overline{s^{-1} B} = \delta_s^{-1} \overline{B}$.
\end{lemma}

\begin{proof}
\uses{lemma_ultrafilter_mult_with_one_principal}
    Each line is equivalent to the one following it.
    \begin{enumerate}
        \item $p \in \overline{s^{-1} B}$
        \item $s^{-1} B \in p$
        \begin{itemize}
            \item By \cref{lemma_ultrafilter_mult_with_one_principal}
        \end{itemize}
        \item $B \in \delta_s p$
        \item $\delta_s p \in \overline{B}$
        \item $p \in \delta_s^{-1} \overline{B}$
    \end{enumerate}
\end{proof}

\begin{lemma}
    \label{lemma_ru_is_r_for_idempotent}
    Let $L \subseteq \beta S$ be a minimal left ideal, and let $u \in L$  be idempotent.  For all $p \in L$, $pu = p$.
\end{lemma}

\begin{proof}
    Let $p \in L$.  Note that $Lu$ is a left ideal contained in $L$.  By minimality, $Lu = L$.  Therefore, there exists $q \in L$ such that $p = qu$.  Then $pu = quu = qu = p$, as was to be shown.
\end{proof}

\subsection{Ultrafilters as the phase space of an $S$-system}

\begin{theorem}
    \label{thm_ultrafilters_as_system}
    Let $S$ be a semigroup.  With left multiplication by elements of $S$ (via principal ultrafilters) and the topology described above, $\beta S$ is an $S$-system.
\end{theorem}

\begin{proof}
\uses{lemma_left_multiplication_is_continuous}
    $\beta S$ is a compact Hausdorff space.  The semigroup $S$ acts on $\beta S$ by multiplication on the left, which, by \cref{lemma_left_multiplication_is_continuous}, is continuous.
\end{proof}

\begin{theorem}[Subsystems in $\beta S$]
\label{thm_beta_S_subsystem_characterization}
    Let $S$ be a semigroup.  The $S$-subsystems of $\beta S$ are precisely the closed left ideals of $\beta S$.
\end{theorem}

\begin{proof}
\uses{thm_ultra_right_mult_continuous,lemma_image_closure_equals_closure_image}
    Let $Z$ be an $S$-subsystem of $\beta S$.  Then $SZ \subseteq Z$.  Let $z \in Z$.  By \cref{thm_ultra_right_mult_continuous}, $\rho_z: \beta S \to \beta S$ is continuous and $\rho_z S \subseteq Z$.  By \cref{lemma_image_closure_equals_closure_image},
    \[\rho_z(\beta S) = \rho_z(\overline{S}) = \overline{\rho_z S} \subseteq \overline{Z} = Z,\]
    whereby $(\beta S)z \subseteq Z$.  Since $z$ was arbitrary, we have that $(\beta S)Z \subseteq Z$, meaning that $Z$ is a left ideal.  Since $Z$ is closed, we see that $Z$ is a closed left ideal, as desired.

    Conversely, suppose that $L$ is a closed left ideal.  Since $(\beta S)L \subseteq L$, we have in particular that $SL \subseteq L$.  So, $L$ is non-empty, closed, and $S$-invariant.  Therefore, it is an $S$-subsystem of $\beta S$.
\end{proof}

\begin{theorem}[Minimal subsystems in $\beta S$]
\label{thm_minimal_beta_S_subsystem_characterization}
    Let $S$ be a semigroup.  The minimal $S$-subsystems of $\beta S$ are precisely the minimal left ideals of $\beta S$.
\end{theorem}

\begin{proof}
\uses{thm_beta_S_subsystem_characterization,lemma_left_ideal_contains_closed_left_ideal}
    Let $Z$ be a minimal $S$-subsystem of $\beta S$.  By \cref{thm_beta_S_subsystem_characterization}, $Z$ is a left ideal of $\beta S$.  Let $L \subseteq Z$ be a left ideal of $S$ contained in $Z$.  By \cref{lemma_left_ideal_contains_closed_left_ideal}, there exists a closed left ideal $L'$ of $S$ contained in $L$.  By \cref{thm_beta_S_subsystem_characterization}, $L'$ is an $S$-subsystem of $\beta S$.  Since $Z$ is a minimal $S$-subsystem, $L' = L = Z$.  Therefore, we have shown that $Z$ is a minimal left ideal of $\beta S$.

    Conversely, suppose $L$ is a minimal left ideal of $S$. By \cref{thm_beta_S_subsystem_characterization}, $L$ is an $S$-subsystem of $\beta S$.  To see that it is minimal, suppose $Z \subseteq L$ is an $S$-subsystem of $L$. By \cref{thm_beta_S_subsystem_characterization}, $Z$ is a left ideal of $S$ contained in $L$.  Since $L$ is minimal, $Z = L$, showing that $L$ is a minimal $S$-subsystem of $\beta S$.
\end{proof}

\begin{theorem}[{cf. \cite[Lemma 19.6]{hindman_strauss_book_2012}}]
\label{thm_minimal_ultrafilter_iff_unif_rec}
    Let $S$ be a semigroup.  An ultrafilter is minimal if and only if it is $S$-uniformly recurrent.
\end{theorem}

\begin{proof}
\uses{thm_minimal_beta_S_subsystem_characterization,lemma_pts_in_minimal_are_ur,lemma_orb_closure_of_ur_is_minimal,def_minimal_ultrafilter}
    Suppose $p \in \beta S$ is minimal.  By definition, it is contained in a minimal left ideal $L$.  By \cref{thm_minimal_beta_S_subsystem_characterization}, $L$ is a minimal $S$-subsystem of $\beta S$. By \cref{lemma_pts_in_minimal_are_ur}, $p$ is $S$-uniformly recurrent.

    Conversely, suppose that $p$ is $S$-uniformly recurrent.  By \cref{lemma_orb_closure_of_ur_is_minimal}, $\overline{Sp}$ is a minimal $S$-subsystem of $\beta S$ containing $p$. By \cref{thm_minimal_beta_S_subsystem_characterization}, $\overline{Sp}$ is a minimal left ideal.  Therefore, $p$ is contained in a minimal left ideal, implying (\cref{def_minimal_ultrafilter}) it is minimal.
\end{proof}

\subsection{Ultrafilters as a semigroup acting on $X$}



In this section, for $f: S \to X$, recall that $\tilde f: \beta S \to X$ is the unique continuous extension of $f$ to $\beta S$.

\begin{definition}[$p-\lim$]
    Let $S$ be a set, $X$ a compact Hausdorff space, and $f: S \to X$.  We write $p-\lim_{s \in S} f(s)$ to mean $\tilde f (p)$.  (Alternative: equivalently, $p-\lim_{s \in S} f(s)$ is the point $x \in X$ with the property that for all non-empty, open $U \ni x$, $\{s \in S : f(s) \in U\} \in p$.)
\end{definition}

\begin{theorem}[$(pq)-\lim$ description]
\label{thm_pq_lim_is_plim_qlim}
    Let $S$ be a semigroup, $X$ a compact Hausdorff space, and $f: S \to X$.  For all $p, q \in \beta S$,
    \[(pq)-\lim_s f(s) = p-\lim_s (q-\lim_t f(st)).\]
\end{theorem}

\begin{proof}
\uses{lemma_left_multiplication_is_continuous,thm_ultra_right_mult_continuous}
    \dgg{These steps are broken into separate results in the lean miniproject on this topic.}
    
    Step 1: for $s \in S$ and $f: S \to X$,
    \[\widetilde{f \circ \lambda_s} = \tilde f \circ \lambda_{\delta_s}.\]
    This needs that multiplication on the left by principal ultrafilters is continuous (\cref{lemma_left_multiplication_is_continuous}).

    Step 2: for $q \in \beta S$ and $f: S \to X$, the extension of the map $s \mapsto \widetilde{f \circ \lambda_s}(q)$ to $\beta S$ is $\tilde f \circ \rho_q$. This requires that multiplication on the right is continuous (\cref{thm_ultra_right_mult_continuous}).

    Step 3: For $p, q \in \beta S$ and $f: S \to X$, the extension of the map $s \mapsto \widetilde{f \circ \lambda_s}(q)$ evaluated at $p$ is equal to $\tilde f(pq)$.  That is to say, $(pq)-\lim_s f(s) = p-\lim_s (q-\lim_t f(st))$.
\end{proof}

\begin{theorem}[Continuous function and $p-\lim$]
\label{thm_continuity_and_plim}
    Let $S$ be a semigroup, $X, Y$ compact Hausdorff spaces, $f: S \to X$, and $\pi: X \to Y$ continuous.  For all $p \in \beta S$, $p-\lim_{s \in S} \pi (f (s)) = \pi (p-\lim_{s \in S} f (s))$.
\end{theorem}

\begin{proof}
    The statement here is that $\widetilde {\pi \circ f} = \pi \circ \tilde f$.  This is easy to see by checking the equality on the dense set of principal ultrafilters.  Indeed, for $s \in S$, $(\pi \circ f)(s) = \pi(f(s))$.
\end{proof}

Notation: For $p \in \beta S$ and $x \in X$, write $px$ for $p-\lim_{s \in S} sx$.

\begin{theorem}[Ultrafilter semigroup action theorem]
\label{thm_ultra_semigroup_action}
    Let $S$ be a semigroup. Let $X$ be an $S$-system.  The map $\beta S \times X \to X$ given by $(p,x) \mapsto p-\lim_{s \in S} sx$ yields an action of $\beta S$ on $X$.
\end{theorem}

\begin{proof}
\uses{thm_pq_lim_is_plim_qlim,thm_continuity_and_plim}
    For $p, q \in \beta S$ and $x \in X$, we want to show that 
    \[(pq) x = p(qx).\]
    To see this, we get by \cref{thm_pq_lim_is_plim_qlim} and using that acting on the left by $s$ is a continuous map $X \to X$(\cref{thm_continuity_and_plim}), we see
    \begin{align*}
        (pq) x &= (pq)-\lim_s sx\\
        &= p-\lim_s (q-\lim_t (st)x ) \\
        &= p-\lim_s (q-\lim_t s(tx)) \\
        &= p-\lim_s s(q-\lim_t tx) \\
        &= p-\lim_s s (qx) \\
        &= p(qx).
    \end{align*}
\end{proof}

\begin{theorem}
    \label{thm_ultrafilter_action_with_fixed_x_is_continuous}
    Let $S$ be a semigroup, $X$ be an $S$-system, and $x \in X$.  The map $\beta S \to X$ defined by $p \mapsto px$ is continuous.
\end{theorem}

\begin{proof}
    By definition, the map $\beta S \to X$ defined by $p \mapsto px$ is the lift of the map $S \to X$ defined by $s \mapsto sx$.  Therefore, it is continuous.
\end{proof}

\begin{theorem}
    \label{thm_ultra_factor_map}
    Let $S$ be a semigroup, $X$ be an $S$-system, and $x \in X$.  If $\overline{Sx} = X$, then the map $\beta S \to X$ defined by $p \mapsto px$ is a factor map of $S$-systems.
\end{theorem}

\begin{proof}
\uses{thm_ultrafilter_action_with_fixed_x_is_continuous}
    By \cref{thm_ultrafilter_action_with_fixed_x_is_continuous}, the map is continuous.  Since $\overline{Sx} = X$, we have that $p \mapsto px$, the lift of $s \mapsto sx$, is surjective.
\end{proof}

\begin{theorem}
    \label{thm_ultra_point_in_orb_closure}
    Let $S$ be a semigroup, $X$ be an $S$-system, $x \in X$, and $p \in \beta S$. Then $px \in \overline{Sx}$.
\end{theorem}

\begin{proof}
    Recall that the map $\beta S \to X$ defined by $p \mapsto px$ is the lift of the map $S \to X$ defined by $s \mapsto sx$.  Fix $x \in X$.  For all $s \in S$, $sx \in \overline{Sx}$.  Therefore, for all $p \in \beta S$, $px \in \overline{Sx}$.
\end{proof}

\begin{theorem}[Ultrafilter action through factors]
\label{thm_ultra_semigroup_action_thru_factor}
    Let $X$ and $Y$ be $S$-systems, and let $\pi: X \to Y$ be a factor map.  For all $p \in \beta S$, $p \pi = \pi p$ as maps from $X \to Y$.
\end{theorem}

\begin{proof}
\uses{thm_continuity_and_plim}
    Let $p \in \beta S$ and $x \in X$.  By swapping $p-\lim$ and $\pi$ (\cref{thm_continuity_and_plim}), we see that
    \[p \pi x = p-\lim_{s \in S} s \pi x = p-\lim_{s \in S} \pi s x = \pi p-\lim_{s \in S} s x = \pi p x,\]
    as desired.
\end{proof}

\begin{theorem}
\label{lemma_plim_in_open_set_return_time}
    Let $S$ be a semigroup and $X$ be an $S$-system.  Let $x \in X$, $U \subseteq X$ be open, and $p \in \beta S$.
    \begin{enumerate}
        \item If $px \in U$, then $R(x,U) \in p$.
        \item If $R(x,U) \in p$, then $px \in \overline{U}$.
    \end{enumerate}
\end{theorem}

\begin{proof}
    Both of these statements follow immediately from the definition of the filter limit.
\end{proof}

\begin{lemma}[Surjectivity lemma]
\label{lemma_ultrafilter_surjective_action}
Let $X$ be a minimal $S$-system and $L \subseteq \beta S$ be a minimal left ideal.  For all $x, y \in X$, there exists $p \in L$ such that $px = y$.
\end{lemma}

\begin{proof}
\uses{thm_ultra_factor_map,thm_minimal_beta_S_subsystem_characterization,lemma_img_of_inv_is_inv}
    Fix $x \in X$. By the minimality of the $S$-action, $\overline{Sx} = X$.  By \cref{thm_ultra_factor_map}, the map $\pi: \beta S \to X$ given by $p \mapsto px$ is a factor map.  By \cref{thm_minimal_beta_S_subsystem_characterization}, $L$ is an $S$-minimal subsystem of $\beta S$.  By \cref{lemma_img_of_inv_is_inv}, $\pi L$ is an $S$-subsystem of $X$.  But $X$ is minimal, so $\pi L = X$.  Therefore, there exists $p \in L$ such that $px = y$.
\end{proof}

\begin{lemma}[Idempotent ultrafilter lemma]
\label{lemma_exists_idempotent_px_is_x}
Let $S$ be a semigroup and $X$ be a minimal $S$-system.  Let $L \subseteq \beta S$ be a minimal left ideal.  For all $x \in X$, there exists an idempotent $p \in L$ such that $px = x$.
\end{lemma}

\begin{proof}
\uses{lemma_ultrafilter_surjective_action,thm_ultrafilter_action_with_fixed_x_is_continuous,lemma_minimal_left_ideal_compact,thm_ultrafilters_are_compact_right_top,thm_ultra_semigroup_action,exists_idempotent_in_compact_subsemigroup}
    Fix $x \in X$. Let $A = \{p \in \beta S : px = x\}$.  By \cref{lemma_ultrafilter_surjective_action}, $A \cap L$ is non-empty. Since $\{x\}$ is closed and $p \mapsto px$ is continuous (\cref{thm_ultrafilter_action_with_fixed_x_is_continuous}), the set $A$ is closed.  Since $L$ is closed (\cref{lemma_minimal_left_ideal_compact} and \cref{thm_ultrafilters_are_compact_right_top}), $A \cap L$ is closed. By \cref{thm_ultra_semigroup_action}, if $p, q \in A$, then $pq \in A$.  Since $L$ is a left ideal, if $p, q \in L$, then $pq \in L$.  Therefore, $A \cap L$ is a closed, hence compact, subsemigroup of $\beta S$. Apply \cref{exists_idempotent_in_compact_subsemigroup}.
\end{proof}

\begin{lemma}[Points $x$ and $px$ are proximal]
    \label{lemma_idempotent_implies_proximal}
    Let $X$ be an $S$-system, $x \in X$, and $p \in \beta S$ an idempotent ultrafilter.  The points $x$ and $px$ are proximal.
\end{lemma}

\begin{proof}
\uses{thm_ultra_semigroup_action,def_proximal}
    Let $\alpha \subseteq X^2$ be an entourage.  By the semigroup action theorem (\cref{thm_ultra_semigroup_action}), 
    \[p(x,px) = (px,ppx) = (px,px) \in \Delta \subseteq \alpha.\]
    Therefore, by the definition of proximal (\cref{def_proximal}) $p-\lim$, there exists (many) $s \in S$ such that $(sx,spx) \in \alpha$, as was to be shown.
\end{proof}

\begin{lemma}[$px$ is uniformly recurrent]
    \label{lemma_minimal_implies_uniformly_recurrent}
    Let $S$ be a semigroup, $X$ be an $S$-system, $x \in X$, and $p \in \beta S$ be a minimal ultrafilter.  The point $px$ is $S$-uniformly recurrent.
\end{lemma}

\begin{proof}
\uses{thm_ultrafilters_as_system,thm_ultra_semigroup_action,thm_minimal_ultrafilter_iff_unif_rec}
    Let $U \ni px$.  Choose an open set $V \ni px$ such that $\overline{V} \subseteq U$.  By the definition of $p-\lim$, $p \in \overline{R(x,V)}$.  By \cref{thm_ultrafilters_as_system}, we can treat $\beta S$ as an $S$-system. We claim that
    \[R(p,\overline{R(x,V)}) \subseteq R(px,U).\]
    Indeed, note first that $s \in R(p,\overline{R(x,V)})$ implies $sp \in \overline{R(x,V)}$ implies $R(x,V) \in sp$, which by the definition of $p-\lim$ implies $(sp)x \in \overline{V} \subseteq U$.  By the ultrafilter action theorem (\cref{thm_ultra_semigroup_action}), $(sp)x = s (px)$, so $s(px) \in U$, which implies that $s \in R(px,U)$.

    Since $p$ is minimal, by \cref{thm_minimal_ultrafilter_iff_unif_rec}, $p$ is $S$-uniformly recurrent.  Therefore, $R(p,\overline{R(x,V)})$ is syndetic. Therefore, so is $R(px,U)$, as was to be shown.
\end{proof}

\begin{lemma}[Technical idempotent lifting lemma]
\label{lemma_ultrafilter_idempotent_lift}
    Let $X$ be a minimal $S$-system and $x \in X$.  Let $L \subseteq \beta S$ be a minimal left ideal and $p \in L$.  If $p x = x$, there exists $q \in L$ such that $q x = x$ and $pq$ is idempotent.
\end{lemma}

\begin{proof}
\uses{thm_ultrafilters_as_system,thm_minimal_beta_S_subsystem_characterization,lemma_exists_idempotent_px_is_x,lemma_ultrafilter_surjective_action,thm_ultra_semigroup_action,lemma_ru_is_r_for_idempotent}
    By \cref{thm_ultrafilters_as_system}, we can treat $\beta S$ as an $S$-system. By \cref{thm_minimal_beta_S_subsystem_characterization}, $L$ is a minimal $S$-system. By \cref{lemma_exists_idempotent_px_is_x}, there exists an idempotent $u \in L$ so that $up = p$.  By \cref{lemma_ultrafilter_surjective_action} (twice), there exist $q, r \in L$ such that $qp = u$ and $rq = u$. By the semigroup action of $\beta S$ on $X$ (\cref{thm_ultra_semigroup_action}),
    \[qx = q (p x) = (qp) x = ux = u (px) = (up) x = px = x.\]
    By \cref{lemma_ru_is_r_for_idempotent}, $ru = r$.  Therefore, we also have
    \[pq = (up)q = u (pq) = (rq) (pq) = r (qp) q = ruq = (ru) q = rq = u,\]
    as desired.
\end{proof}

\begin{lemma}
\label{lemma_pr_family_concentration}
    Let $X$ be an $S$-system, $x \in X$, and $U \subseteq X$ be closed.  Let $\family$ be a partition regular family of subsets of $S$. If $R(x,U) \in \family$, then there exists $y \in U$ such that for all non-empty, open $V \ni y$ open and non-empty, $R(x,V) \in \family$.
\end{lemma}

\begin{proof}
    \dgg{This argument can be shortened considerably using filter limits, which are already part of mathlib.  The family $\family^*$ is a filter and $U$ is a closed set belonging to the pushforward of $\family^*$ under the action map through the point $x$.  By compactness, there is a cluster point $y$ of the pushforward of $\family^*$.  This says exactly that for all non-empty, open $V \ni y$, $R(x,V) \in \family$.}
    We prove the contrapositive.  Suppose for all $y \in U$, there exists non-empty, open $V_y \ni y$ such that $R(x,V_y) \not\in \family$.  By compactness of $U$, there exists a finite set $F \subseteq X$ such that $U \subseteq \cup_{y \in F} V_y$. Then
    \[R(x,U) \subseteq R(x,\cup_{y \in F} V_y) = \cup_{y \in F} R(x,V_y).\]
    Since no $R(x,V_y)$ is in $\family$, we have that $\cup_{y \in F} R(x,V_y) \not\in \family$, and hence that $R(x,U) \not\in \family$, as desired.
\end{proof}



\subsection{Syndetic and thick sets}

\begin{definition}
Let $(S,\cdot)$ be a semigroup and $A \subseteq S$.
\begin{enumerate}
    \item The set $A$ is syndetic if there exists a finite set $F \subseteq S$ such that $S = \cup_{f \in F} f^{-1} A$.

    \item The set $A$ is thick if for all finite $F \subseteq S$, there exists $s \in S$ such that $Fs \subseteq A$.
\end{enumerate}
\end{definition}

    $\syndetic = \thick^*$
\end{lemma}

\begin{lemma}
    \label{lemma_image_of_syndetic}
    If $\varphi: G \to H$ is a surjective semigroup homomorphism and $A \subseteq G$ is syndetic in $G$, then $\varphi A$ is syndetic in $H$.
\end{lemma}

\begin{proof}[Proof sketch]
    Since $A \subseteq G$ is syndetic, there exists a finite set $F \subseteq G$ such that for all $g \in G$, there exists $f \in F$ such that $fg \in A$.

    We claim that $\varphi F$ has the property that for all $h \in H$, there exists $f \in F$ such that $\varphi(f)h \in \varphi A$.  Indeed, let $h \in H$.  Write $h = \varphi g$ for some $g \in G$.  There is $f \in F$ such that $fg \in A$.  Apply $\varphi$ to see that $\varphi(f) h \in \varphi A$, as desired.
\end{proof}

\end{lemma}

\end{proof}

\begin{lemma}
    \label{lemma_thick_shift_intersections}
    Let $S$ be a commutative semigroup and $H \subseteq S$ be thick. For all finite $K \subseteq S$, the set $\cap_{k \in K} kH$ is thick.
\end{lemma}

\dgg{An older version of this lemma for groups is in the comments.}

\begin{proof}
\uses{lemma_thick_inverse_shift_intersections}
    Let $K \subseteq S$ be finite.  For $k \in K$, let $p_k$ be the product of all the elements of $K$ except $k$, and let $p$ be the product of all elements of $K$.  By \cref{lemma_thick_inverse_shift_intersections}, the set $\cap_{k \in K} p_k^{-1} H$ is thick.
    
    To see that $\cap_{k \in F} kH$ is thick, let $F \subseteq S$ finite.  Since $\cap_{k \in K} p_k^{-1} H$ is thick, there exists $s \in S$ such that $Fs \subseteq \cap_{k \in K} p_k^{-1} H$.  Therefore,
    \[Fps = p F s \subseteq p \cap_{k \in K} p_k^{-1} H \subseteq \cap_{k \in K} p p_k^{-1} H \subseteq \cap_{k \in K} k H,\]
    as was to be shown.
\end{proof}

\begin{lemma}
    \label{lemma_thick_closure_contains_ideal}
    Let $S$ be a semigroup and $H \subseteq S$ be thick.  The set $\overline{H}$ contains a minimal left ideal.
\end{lemma}

\begin{proof}
\uses{lemma_thick_inverse_shift_intersections,thm_ultrafilters_are_compact_right_top,lemma_pull_out_inverse_mult_in_basis_set,thm_ultra_right_mult_continuous,lemma_image_closure_equals_closure_image,thm_every_left_contains_minimal_left}
    By \cref{lemma_thick_inverse_shift_intersections}, the set
    \[\big\{ \bigcap_{f \in F} \overline{f^{-1} H} \mid F \subseteq G \text{ finite, non-empty}\big\}\]
    has the finite intersection property.  Since $\beta S$ is compact (\cref{thm_ultrafilters_are_compact_right_top}), the set
    \[L \defeq \bigcap_{g \in S} \overline{g^{-1}H}\]
    is non-empty.
    
    We will show that $L$ is a left ideal.  Let $p \in L$.  We will show first that for all $s \in S$, $\delta_s p \in L$.  Let $s \in S$, $g \in S$. We must show that $\delta_s p \in \overline{g^{-1} H}$.  By the definition of $L$ and \cref{lemma_pull_out_inverse_mult_in_basis_set}, we see
    \[\delta_sp \in \delta_s \overline{(gs)^{-1} H} = \delta_s \overline{s^{-1}g^{-1} H} = \delta_s \delta_s^{-1} \overline{g^{-1} H} \subseteq \overline{g^{-1} H},\]
    as desired.  Denoting right multiplication by $p$ by $\rho_p: \beta S \to \beta S$, we have shown that $\rho_p(\{\delta_s : s \in S\}) \subseteq L$.

    By \cref{thm_ultra_right_mult_continuous}, $\rho_p$ is continuous.  Since $L$ is closed, by \cref{lemma_image_closure_equals_closure_image}, we see
    \[\rho_p(\beta S) = \rho_p(\overline{S}) = \overline{\rho_p(\{\delta_s : s \in S\})} \subseteq \overline{L} = L.\]
    In other words: for all $q \in \beta S$, $qL \subseteq L$, as was to be shown.
    
    By \cref{thm_ultrafilters_are_compact_right_top} and \cref{thm_every_left_contains_minimal_left}, the left ideal $L$ contains a minimal left ideal, hence so does $H$.
\end{proof}

\begin{lemma}
    \label{lemma_thick_closure_contains_strong_ultrafilter}
    Let $S$ be a semigroup and $H \subseteq S$ be thick. There exists a minimal idempotent $p \in \overline{H}$ such that for all finite $F \subseteq S$, the set $\cap_{f \in F} f^{-1} H \in p$.
\end{lemma}

\begin{proof}
\uses{lemma_thick_closure_contains_ideal,lemma_exists_idempotent_in_left_ideals}
    From the proof of \cref{lemma_thick_closure_contains_ideal}, there exists a minimal left ideal
    \[L' \subseteq \bigcap_{g \in S} \overline{g^{-1}H}.\]

    By \cref{lemma_exists_idempotent_in_left_ideals}, $L'$ contains an idempotent $p$.  Then $p$ is a minimal, idempotent ultrafilter.  To see that it has the stated property, let $F \subseteq S$ be finite.  Then
    \[p \in L' \subseteq \bigcap_{f \in F} \overline{f^{-1} H},\]
    which means that $\cap_{f \in F} f^{-1} H \in p$, as desired.
\end{proof}

\begin{lemma}
\label{lemma_preimage_intersection_of_thick_is_thick}
    Let $S$ be a commutative semigroup and $B \subseteq S$ be thick.  For all finite $K \subseteq S$, the set
    \[\bigcap_{k \in K} k^{-1} B\]
    is thick.
\end{lemma}

\begin{proof}
    Let $K \subseteq S$ be finite.  Let $F \subseteq S$ be finite.  The set $\cup_{k \in K} kF$ is finite.  Since $B$ is thick, there exists $s \in S$ such that $(\cup_{k \in K} kF)s \subseteq B$.  Therefore, for all $k \in K$, $kFs \subseteq B$.  That is: $F s \subseteq k^{-1}B$.  In other words, $F s \subseteq \cap_{k \in K}$, as was to be shown.
\end{proof}

\subsection{Delta sets}

\begin{definition}
Let $(S,\cdot)$ be a semigroup and $A \subseteq S$.
\begin{enumerate}
    \item The set $A$ is a $\Delta$ set (Hindman-Strauss call this more generally a $Q$-set, where $Q$ is for quotient) if there exists a sequence $(s_n)_n \subseteq S$ such that for all $1 \leq m < n$, $s_n \in s_m A$.

    \item The set $A$ is a $\Delta_0$ set if for all $N \in \N$, there exists a sequence $(s_n)_{n=1}^N \subseteq S$ such that for all $1 \leq m < n \leq N$, $s_n \in s_m A$.
\end{enumerate}
\end{definition}

\begin{theorem}
    \label{thm_delta_in_deltazero}
    Let $S$ be a semigroup.  $\Delta \subseteq \Delta_0$
\end{theorem}

\begin{proof}
    This is immediate from the definitions.
\end{proof}

\subsection{Bohr sets}

\dgg{In the definition below, we require the system to be minimal.  We can drop that assumption if all points are uniformly recurrent for an equicontinuous semigroup action.  This is true for group actions, but may not be true any longer for semigroup actions.}

\begin{definition}
Let $(S,\cdot)$ be a semigroup and $A \subseteq S$.
\begin{enumerate}
    \item The set $A$ is a $\dcsyndeticbohr$ set if there exists a minimal equicontinuous $S$-system $X$, a point $x \in X$, and a non-empty, open set $U \subseteq X$ such that $R(x,U) \subseteq A$.

    \item The set $A$ is a $\dcthickbohr$ set (that is, a set of Bohr recurrence for $S$ actions) if for all minimal equicontinuous $S$-systems $X$, all $x \in X$, and all $U \subseteq X$ containing $x$, $A \cap R(x,U) \neq \emptyset$.
\end{enumerate}
\end{definition}

\begin{lemma}
    \label{lemma_dcsyndeticbohr_dcthickbohr_dual}
    $\dcsyndeticbohr = \dcthickbohr^*$.
\end{lemma}

\begin{proof}
    Immediate from definitions.
\end{proof}

\begin{lemma}
    \label{lemma_dcsyndeticbohr_is_filter}
    Suppose $S$ is commutative. $\dcsyndeticbohr$ is a filter.
\end{lemma}

\begin{proof}
\uses{lemma_product_of_equi_is_equi,lemma_subsys_of_equi_is_equi,thm_every_system_has_minimal_subsystem,lemma_minimal_equi_implies_homogeneous,lemma_return_time_sets_in_prod_system}
    Let $A, B \in \dcsyndeticbohr$.  Then $A = R(x,U)$ and $B = R(y,V)$ in minimal equicontinuous $S$-systems $X$ and $Y$.  By \cref{lemma_product_of_equi_is_equi}, the $S$-system $X \times Y$ is equicontinuous.  By \cref{lemma_subsys_of_equi_is_equi}, subsystems of equicontinuous systems are equicontinuous.  So, it suffices to show that the point $(x,y)$ belongs to a minimal subsystem of $X \times Y$.

    By \cref{thm_every_system_has_minimal_subsystem}, there exists an $S$-minimal subsystem $Z \subseteq X \times Y$.  Since $\pi_2 Z = Y$, there exists $(w,y) \in Z$.  By \cref{lemma_minimal_equi_implies_homogeneous}, there exists a factor map $\varphi: X \to X$ such that $\varphi(x) = w$.  Apply $\varphi \otimes \text{Id}$ to $Z$ to get a minimal subsystem of $X \times Y$ containing $(x,y)$. By product system visit time algebra (\cref{lemma_return_time_sets_in_prod_system}), $A \cap B = R((x,y), U \times V)$, whereby $A \cap B \in \dcsyndeticbohr$.
\end{proof}

\begin{lemma}
    \label{lemma_dcthickbohr_is_pr}
    $\dcthickbohr$ is partition regular.
\end{lemma}

\begin{proof}
\uses{lemma_dcsyndeticbohr_dcthickbohr_dual,lemma_dcsyndeticbohr_is_filter,thm_family_pr_iff_dual_is_filter}
    That $\dcthickbohr$ is partition regular follows immediately from \cref{lemma_dcsyndeticbohr_dcthickbohr_dual}, \cref{lemma_dcsyndeticbohr_is_filter}, and \cref{thm_family_pr_iff_dual_is_filter}.
\end{proof}

\begin{theorem}
    \label{thm_delta_zero_set_of_bohr_rec}
    For $S$ a commutative semigroup, $\Delta_0 \subseteq \dcthickbohr$.
\end{theorem}

\begin{proof}
\uses{lemma_minimal_comm_action_implies_surjective}
    Suppose $A \in \Delta_0$ and we want to show that $A \cap R(x,U) \neq \emptyset$, where $X$ is a minimal, equicontinuous $S$-system, $x \in U \subseteq X$.  Since $X$ is compact, Hausdorff, it has a unique uniform structure.  Thus, there exists an entourage $\beta \subseteq X^2$ such that $\beta[x] \subseteq U$.  By equicontinuity, there exists an entourage $\alpha \subseteq X^2$ such that for all $s \in S$, $(s \otimes s) \alpha \subseteq \beta$.

    Let $\gamma$ be an entourage such that $\gamma \circ \gamma \subseteq \alpha$. But replacing $\gamma$ with $\gamma \cap \gamma^{-1}$, we can assume $\gamma$ is symmetric (i.e. $(x, y) \in \gamma$ if and only if $(y, x) \in \gamma)$.
    
    Note that $\Delta(X) \subseteq \gamma$, whereby
    \[
        X = \bigcup_{x \in X} \gamma[x].
    \]
    By compactness, there exists a finite set $Z \subseteq X$ such that
    \[
        X = \bigcup_{z \in Z} \gamma[z].
    \]
    Put $N = |Z|+1$.  By the assumption that $A$ is a $\Delta_0$ set, there exists $s_1, \ldots, s_N \in S$ such that for all $1 \leq m < n \leq N$, $s_n \in s_m A$.
    
    By \cref{lemma_minimal_comm_action_implies_surjective}, the elements $s_1, \ldots, s_N$ act surjectively on $X$.  Therefore, there exists $y_1, \ldots, y_N \in X$ such that $s_i y_i = x$.
    
    Consider the points $y_1, \ldots, y_N$.  Since $N > |Z|$, by pigeonhole principle, there are $1 \leq m < n \leq N$ and $z \in Z$ such that $(y_n, y_m) \in \gamma[z] \times \gamma[z]$. It follows that $(z, y_n), (z, y_m) \in \gamma$. Since $\gamma$ is symmetric, $(y_n, z) \in \gamma$. Since $\gamma \circ \gamma \subseteq \alpha$, we have $(y_n, y_m) \in \alpha$.
    
    Since $s_n \in s_m A$, there exist $a \in A$ such that $s_n = s_m a$.
    Act on the product by $s_n$ to see that
    \[
        (x,a x) = (x, a s_m y_m) = (x, s_n y_m) = (s_n \otimes s_n) (y_n, y_m) \in (s_n \otimes s_n) \alpha \subseteq \beta,
    \]
    whereby $a x \in \beta[x]$. Since $\beta[x] \subseteq U$, we have $a \in R(x,U)$.  Note that $a \in A$ and so $A \cap R(x,U) \neq \emptyset$, as was to be shown.
\end{proof}

\begin{remark}
The proof would be a straight-forward application of the pigeonhole principle if $S$ were a group and the phase spaces of the systems we consider were metrizable. In the general setting we are in, without access to inverses in $S$ and even not assuming cancellativity of $S$, more work is required before the pigeonhole principle is applied.
\end{remark}

\begin{theorem}

    Let $X$ be a minimal $S$-system and $x, y \in X$.  The points $(x,y,y)$ and $(y,y,x)$ are $S^2$-uniformly recurrent.
\end{theorem}

\begin{proof}

Since $X$ is a minimal $S$-system, by \cref{lem_minimal_h_from_g}, $X^2$ is a minimal $S^2$-system.  By \cref{lemma_exists_idempotent_px_is_x}, there exists a minimal, idempotent $p \in \beta S^2$ such that $p(x,x) = (x,x)$.  We see by \cref{lemma_ultrafilter_action_on_X3} that $p(x,x,x) = (x,x,p x)$.

Since $p$ is minimal, by \cref{lemma_minimal_implies_uniformly_recurrent}, the point $p(x,x,x) = (x,x,px)$ is $S^2$-uniformly recurrent.  Since $p$ is idempotent, by \cref{lemma_idempotent_implies_proximal}, the points $x$ and $px$ are $S^2$-proximal.

By \cref{lemma_prox_in_h_implies_in_g}, the points $x$ and $px$ are $S$-proximal.  It follows by \cref{theorem_proximal_and_minimal_implies_orbit_closure_contains_diagonal} that $(x,x) \in \overline{S(x,px)}$. By \cref{lemma_S_orb_closure_contained_in_S2_orbit_closure_v3}, $(x,x,x) \in \overline{S^2(x,x,px)}$.  By \cref{lemma_in_orb_clos_of_ur_implies_ur}, all points in $\overline{S^2(x,x,px)}$ are $S^2$-uniformly recurrent, whereby $(x,x,x)$ is $S^2$-uniformly recurrent.

Since $X$ is $S$-minimal, $(y,y) \in \overline{S(x,x)}$. Since $(x,x,x)$ is $S^2$-uniformly recurrent, it follows by \cref{lemma_S_orb_closure_contained_in_S2_orbit_closure_v2} and \cref{lemma_S_orb_closure_contained_in_S2_orbit_closure_v3} that $(y,y,x)$ and $(x,y,y)$ belong to $\overline{S^2(x,x,x)}$. By \cref{lemma_in_orb_clos_of_ur_implies_ur}, all points in $\overline{S^2(x,x,x)}$ are $S^2$-uniformly recurrent, whereby $(y,y,x)$ and $(x,y,y)$ are $S^2$-uniformly recurrent, as desired.
\end{proof}

\begin{theorem}
\label{thm_rp_implies_same_minimal_component}
\label{theorem_cube_space_orbit_implies_rp}
    Let $X$ be a minimal $S$-system and $x, y \in X$. Consider $X^3$ as an $S^2$-system as described above. The point $(x,y) \in \RP$ if and only if $(x,y,y) \in \overline{S^2(y,y,y)}$.
\end{theorem}

\begin{proof}
\uses{lemma_rp_implies_net_iff,lemma_ultrafilter_surjective_action,lemma_ultrafilter_action_on_X3,thm_ultrafilter_action_with_fixed_x_is_continuous,lemma_ultrafilter_idempotent_lift,thm_ultra_point_in_orb_closure,thm_xxx_is_trtl_unif_rec_v3,lemma_S_orb_closure_contained_in_S2_orbit_closure_v2,theorem_orbit_closure_transitivity,eqn_orb_closure_eqn_1,lemma_in_orb_clos_of_ur_implies_ur,lemma_prox_in_h_implies_in_g,lemma_S_orb_closure_contained_in_S2_orbit_closure_v3}
    Suppose $(x,y) \in \RP$. By \cref{lemma_rp_implies_net_iff}, there exists $w \in X$ and a net $(a_n, b_n, s_n)_n \subseteq X \times X \times S$ such that
    \[(a_n, b_n, s_n a_n, s_n b_n) \to (w, w, x, y).\]

    Fix a minimal left ideal $L \subseteq \beta S^2$.  Since $X^2$ is a minimal $S^2$-system, by \cref{lemma_ultrafilter_surjective_action}, there exists $p_n \in L$ such that $p_n (w,w) = (a_n, b_n)$.  By \cref{lemma_ultrafilter_action_on_X3}, $p_n (w,w,y) = (a_n, b_n, p_n y)$.
    
    Since $L$ is compact, the net $(p_n)_n$ in $L$ has an accumulation point $p \in L$.  Pass to a convergent subnet so that
    \[(a_m, b_m, s_m a_m, s_m b_m, p_m) \to (w, w, x, y, p).\]
    By \cref{thm_ultrafilter_action_with_fixed_x_is_continuous}, the map $\beta S^2 \to X^2$ given by $q \mapsto q(w,w)$ is continuous.  Since $p_m \to p$, we have that $p_m(w,w) \to p(w,w)$.  But $p_n(w,w) = (a_n,b_n)$ and $(a_n,b_n) \to (w,w)$, so we see that $p (w,w) = (w,w)$.

    By \cref{lemma_ultrafilter_idempotent_lift}, there exists $q \in L$ such that $q (w,w) = (w,w)$ and $pq$ is idempotent.  Since the map $\beta S^2 \to X$ given by $r \mapsto rqy$ is continuous (\cref{thm_ultrafilter_action_with_fixed_x_is_continuous}) and $p_m \to p$, we see that $p_m q y \to p q y$. Consequently:
    \begin{align}
        \label{eqn_orb_closure_eqn_1}
        T_{L,s_m} p_m q (w,w,y) = T_{L,g_m} p_m (w,w, q y) = (s_ma_m, s_mb_m, p_m q y) \to (x, y, p q y).
    \end{align}
    Since $p_mq \in L \subseteq \beta S^2$, we have by \cref{thm_ultra_point_in_orb_closure} that $(a_m, b_m, p_m q y) \in \overline{S^2 (w,w,y)}$.
    Since $(w,w,y)$ is $S^2$-uniformly recurrent (by \cref{thm_xxx_is_trtl_unif_rec_v3}) and $(s_ma_m, s_mb_m) \in \overline{S(a_m,b_m)}$, we have by \cref{lemma_S_orb_closure_contained_in_S2_orbit_closure_v2} that $(s_ma_m, s_mb_m, p_m q y) \in \overline{S^2 (a_m,b_m,p_mqy)}$.  By transitivity (\cref{theorem_orbit_closure_transitivity}), we get that
    \[(s_ma_m, s_mb_m, p_m q y) \in \overline{S^2 (w,w,y)}.\]
    Since $\overline{S^2 (w,w,y)}$ is closed, we conclude from \cref{eqn_orb_closure_eqn_1} that $(x, y, p q y) \in \overline{S^2 (w,w,y)}$.
    
    Since $X$ is a minimal $S$-system, by the definition of minimal and the diagonal action, $(w,w) \in \overline{S(y,y)}$.  Since $(y,y,y)$ is $S^2$-uniformly recurrent (\cref{thm_xxx_is_trtl_unif_rec_v3}), by \cref{lemma_S_orb_closure_contained_in_S2_orbit_closure_v2}, we have that $(w,w,y) \in \overline{S^2(y,y,y)}$. By transitivity (\cref{theorem_orbit_closure_transitivity}), we get that
    \[(x, y, pq y) \in \overline{S^2(y,y,y)}.\]
    Note that since $(y,y,y)$ is $S^2$-uniformly recurrent (\cref{thm_xxx_is_trtl_unif_rec_v3}), by \cref{lemma_in_orb_clos_of_ur_implies_ur}, we have that $(x,y, pq y)$ is $S^2$-uniformly recurrent.

By transitivity (\cref{theorem_orbit_closure_transitivity}), we have that
    \[(x, y, y) \in \overline{S^2 (y,y,y)},\]

\subsection{RP is an equivalence relation}

\begin{theorem}
\label{theorem_rp_is_transitive}
    Let $S$ be a commutative semigroup, and let $X$ be a minimal $S$-system.  The $\RP$ relation is transitive.
\end{theorem}

\begin{proof}
\uses{thm_rp_implies_same_minimal_component,thm_xxx_is_trtl_unif_rec_v3,lemma_S_orb_closure_contained_in_S2_orbit_closure_v3,theorem_orbit_closure_transitivity,theorem_cube_space_orbit_implies_rp}
    Let $(x,y), (y,z) \in \RP$.  Since $(x,y) \in \RP$, by \cref{thm_rp_implies_same_minimal_component}, we have that $(x,y,y) \in \overline{S^2(y,y,y)}$.  By minimality, $(z,z) \in \overline{S(y,y)}$.  Since $(x,y,y)$ is $S^2$-uniformly recurrent (\cref{thm_xxx_is_trtl_unif_rec_v3}), by \cref{lemma_S_orb_closure_contained_in_S2_orbit_closure_v3}, $(x,z,z) \in \overline{S^2(x,y,y)}$.  Since $(y,z) \in \RP$, by \cref{thm_rp_implies_same_minimal_component}, we have that $(y,z,z) \in \overline{S^2(z,z,z)}$.  By minimality, $(y,y) \in \overline{S(z,z)}$.  Since $(y,z,z)$ is $S^2$-uniformly recurrent (\cref{thm_xxx_is_trtl_unif_rec_v3}), by \cref{lemma_S_orb_closure_contained_in_S2_orbit_closure_v3}, $(y,y,y) \in \overline{S^2(y,z,z)}$.
    
    Recapping the conclusions from the previous paragraph:
    \begin{align*}
        (y,z,z) \in \overline{S^2(z,z,z)}\\
        (y,y,y) \in \overline{S^2(y,z,z)}\\
        (x,y,y) \in \overline{S^2(y,y,y)}\\
        (x,z,z) \in \overline{S^2(x,y,y)}\\
    \end{align*}
    By transitivity (\cref{theorem_orbit_closure_transitivity}), we see that
    \[(x,z,z) \in \overline{S^2(z,z,z)}.\]
    It follows by \cref{theorem_cube_space_orbit_implies_rp} that $(x,z) \in \RP$, as was to be shown.
\end{proof}

\subsection{A set recurrence corollary}

Here is another application of the material in this section. The theorem below is used later to build $\Delta$ sets.  A version of this theorem for abelian group actions can be found already in Veech \cite{veech_1968}.

\begin{theorem}
\label{thm_delta_set_builder_prestep}
    Let $S$ be a commutative semigroup, let $X$ be a minimal $S$-system, and let $x, y \in X$.  If $(x,y) \in \RP$, then for all non-empty, open $U, V \subseteq X$ with $y \in V$, we have that
    \[R(x,U) \cap R(V,U)\]
    is syndetic.
\end{theorem}

\begin{proof}
\uses{lemma_rp_basics,thm_rp_implies_same_minimal_component,thm_xxx_is_trtl_unif_rec_v3,lemma_S_orb_closure_contained_in_S2_orbit_closure_v3,theorem_orbit_closure_transitivity,lemma_return_time_algebra_three,lemma_image_of_syndetic}
    Let $U, V \subseteq X$ be non-empty, open with $y \in V$.  Let $z \in U$.  By \cref{lemma_rp_basics}, $\RP$ is symmetric, so $(y,x) \in \RP$.  By \cref{thm_rp_implies_same_minimal_component}, $(y,x,x) \in \overline{S^2 (x,x,x)}$. By minimality, $(z,z) \in \overline{S(x,x)}$.  Since $(x,x,x)$ is $S^2$-uniformly recurrent (\cref{thm_xxx_is_trtl_unif_rec_v3}), by \cref{lemma_S_orb_closure_contained_in_S2_orbit_closure_v3}, $(y,z,z) \in \overline{S^2(y,x,x)}$.  By transitivity (\cref{theorem_orbit_closure_transitivity}), $(y,z,z) \in \overline{S^2 (x,x,x)}$.
    
    By \cref{thm_xxx_is_trtl_unif_rec_v3}, the point $(x,x,x)$ is $S^2$-uniformly recurrent.  Since $V \times U \times U$ is an open neighborhood of $(y,z,z)$ and $(y,z,z) \in \overline{S^2 (x,x,x)}$, we get that $R_{S^2}((x,x,x), V \times U \times U)$ is syndetic. This is the set of pairs $(g,r) \in S^2$ such that $T_{L,g} T_{R,r} (x,x,x) \in V \times U \times U$ -- that is, such that $r \in R(x,U)$, $g \in R(x,V)$, $rg \in R(x,U)$.  For each such pair $(g,r) \in S^2$, we see by \cref{lemma_return_time_algebra_three} that
    \[r \in R(x,U) \cap (R(x,U) R(x,V)^{-1}) = R(x,U) \cap R(V,U).\]
    Applying $\pi_2: S^2 \to S$, a surjective semigroup homomorphism, to this set of pairs and using \cref{lemma_image_of_syndetic}, we see that the set $R(x,U) \cap R(V,U)$ is syndetic, as desired.
\end{proof}

\begin{definition}
    Let $S$ be a semigroup. A set $A \subseteq S$ is called a \emph{clopen return-time set of a uniformly recurrent point (CRTURP)} if there exists an $S$-system $X$, an $S$-uniformly recurrent point $x \in X$, and a clopen set $U \subseteq X$ such that $A = R(x,U)$.
\end{definition}

\begin{definition}[Symbolic system]
\label{def_symbolic_system}
    Let $S$ be a semigroup.  The symbolic system on $S$ is set $X = \{0,1\}^S$ with the product topology (so that it is compact and Hausdorff) together with the action of $S$ on $X$ given for $s \in S$ and $\omega \in X$ by
    \[\forall \ t \in S, \ (s \omega)(t) = \omega(st).\]
\end{definition}

\begin{definition}[Cylinder set]
    Let $S$ be a set, $s \in S$, and $a \in \{0,1\}$. The cylinder set $[a]_s$ is defined as
    \[\{\omega \in \{0,1\}^{S} : \omega(s) = a\}.\]
\end{definition}

\begin{lemma}
\label{lem:auslander-ellis}
Let $S$ be a commutative semigroup.  Let $A \subseteq S$ be a syndetic set and $H \subseteq S$ be a thick set. There exists a CRTURP set $B \subseteq S$ and a thick set $H' \subseteq H$ such that $A \cap H' = B \cap H'$.
\end{lemma}

\begin{proof}
\uses{lemma_thick_closure_contains_strong_ultrafilter,lemma_minimal_implies_uniformly_recurrent,lemma_unif_rec_implies_CRTURP,lemma_plim_in_open_set_return_time}
    Since $H$ is thick, by \cref{lemma_thick_closure_contains_strong_ultrafilter}, there exists a minimal, idempotent ultrafilter $p \in \overline{H} \subseteq \beta S$ such that for all finite $F \subseteq S$, $\cap_{f \in F} f^{-1} H \in p$. Let $1_B = p 1_{A \cap H}$.  It follows from \cref{lemma_minimal_implies_uniformly_recurrent} that $1_B$ is $S$-uniformly recurrent.  By \cref{lemma_unif_rec_implies_CRTURP}, $B$ is a CRTURP set.  Define $T = \{g \in S \mid 1_B(g) = 1_{A \cap H}(g) \}$.
    
    We claim that the set $H \cap T$ is thick.  Indeed, let $F \subseteq S$ be finite. The set
    \[U \defeq \{(\alpha, \beta) \in (\{0,1\}^S)^2 \mid \forall f \in F, \ \alpha(f) = \beta (f)\} \subseteq (\{0,1\}^S)^2\]
    is, by the definition of the product topology on $\{0,1\}^S$, an open set containing the diagonal.  Since $p = pp$, we see
    \[p 1_B = p p 1_{A \cap H} = p 1_{A \cap H},\]
    we see by \cref{lemma_plim_in_open_set_return_time} that $R_S((1_B,1_{A \cap H}), U) \in p$ in the product space $(\{0,1\}^S)^2$ under the diagonal action.  By the property of $p$, we see that $\bigcap_{f \in F} f^{-1} H \in p$.  Since $p$ is a filter,
    \[\bigcap_{f \in F} f^{-1} H \cap R_S((1_B,1_{A \cap H}), U) \in p.\]
    Choose $g$ in this set.  We see for all $f \in F$, by commutativity, $gf = fg \in H$ and
    \[1_B(gf) = (g1_B)(f) = (g1_{A \cap H})(f) = 1_{A \cap H}(gf).\]
    Therefore, $Fg \subseteq H \cap T$, as desired.
    
    Set $H' = H \cap T$.  Then $H' \subseteq H$ is thick and $1_B$ and $1_{A \cap H}$ agree on $H'$, that is, $A \cap H' = B \cap H'$, as desired.
\end{proof}

\begin{theorem}
\label{thm_lspwf_upgrade}
    Let $S$ be a commutative semigroup. Let $\family$ and $\familytwo$ be families of subsets of $S$. Suppose that for all CRTURP sets $B \subseteq S$ and all thick $H \subseteq S$, if $B \cap H \in \family$, then $B \cap H \in \familytwo$.  Then we have
    \[\syndetic \classcapdual \family \subseteq \syndetic \classcapdual \familytwo.\]
\end{theorem}

% \dgg{Note: I removed the assumption that $\family \neq \mathcal{P}(S)$, since by definition, if $\family = \mathcal{P}(S)$, then $S \classcapdual \family = S$.}

\begin{proof}
\uses{thm_classcapdual_contained_in_intersection,lem:auslander-ellis}
\begin{enumerate}
    \item Let $A \in \syndetic \classcapdual \family$ and $H \subseteq S$ be a thick set. Our goal is to show that $A \cap H \in \familytwo$.

    \item By \cref{thm_classcapdual_contained_in_intersection}, $\syndetic \classcapdual \family \subseteq \syndetic$, so $A \in \syndetic$.

    \item By \cref{lem:auslander-ellis}, there exists a thick set $H' \subseteq H$ and a CRTURP set $B$ such that $A \cap H' = B \cap H'$.

    \item Since $A \in \syndetic \classcapdual \family$ and $H' \in \thick$, we have $A \cap H' \in \family$ (this holds by the definition of $\classcapdual$, even if $\family = \mathcal{P}(S)$), and so $B \cap H' \in \family$. 

    \item Since $B$ is CRTURP, by our assumption, $B \cap H' \in \familytwo$.

    \item Thus $A \cap H' \in \familytwo$.

    \item Since $H' \subseteq H$, we deduce that $A \cap H \in \familytwo$.
    \end{enumerate}
\end{proof}

\subsection{A delta set builder}

\begin{theorem}
\label{thm_delta_from_kronecker}
    Let $S$ be a commutative semigroup, let $X$ be a minimal $S$-system, $(x, y) \in \RP$, and $V \subseteq X$ an open neighborhood of $y$.  For all thick $B \subseteq S$, the set $R(x,V) \cap B \in \Delta$.
\end{theorem}

\begin{proof}
\uses{thm_delta_set_builder_prestep,lemma_synd_thick_dual,lemma_minimal_comm_action_implies_surjective,lemma_return_time_algebra_one,lemma_return_to_intersection,lemma_preimage_intersection_of_thick_is_thick}
    Let $B \subseteq S$ be thick. Put $U_1 = V$ and $B_1 = B$.

    \begin{enumerate}
        \item[Step 1]
        \begin{enumerate}
            \item By \cref{thm_delta_set_builder_prestep}, the set $R(x,U_1) \cap R(V,U_1)$ is syndetic.  By assumption, the set $B_1$ is thick. By \cref{lemma_synd_thick_dual}, the intersection of a thick set and a syndetic set is non-empty.  Therefore, there exists
            \[t_1 \in R(x,U_1) \cap R(V,U_1) \cap B_1.\]
            
            \item Choose $s_1 \in S$ arbitrarily. Since $t_1 \in R(x,V) \cap B$, we see that
            \begin{align}
                \label{eqn_delta_it_one}
                s_2 \defeq s_1t_1 \in s_1 (R(x,V) \cap B).
            \end{align}

            \item Since $t_1 \in R(V,V)$, $V \cap t_1^{-1} V \neq \emptyset$.  Since $s_1$ acts surjectively (\cref{lemma_minimal_comm_action_implies_surjective}),
            \[U_2 \defeq s_1^{-1} V \cap s_2^{-1} V = s_1^{-1} (V \cap t_1^{-1} V) \neq \emptyset.\]
            Note that by \cref{lemma_return_time_algebra_one} and \cref{lemma_return_to_intersection},
            \[R(x,U_2) = s_1^{-1} R(x,V) \cap s_2^{-1} R(x,V).\]
            Put
            \[B_2 \defeq s_1^{-1} B \cap s_2^{-1} B.\]
        \end{enumerate}

        \item[Step 2]
        \begin{enumerate}
            \item By \cref{thm_delta_set_builder_prestep}, the set $R(x,U_2) \cap R(V,U_2)$ is syndetic.  By \cref{lemma_preimage_intersection_of_thick_is_thick}, the set $B_2$ is thick. By \cref{lemma_synd_thick_dual}, the intersection of a thick set and a syndetic set is non-empty.  Therefore, there exists
            \[t_2 \in R(x,U_2) \cap R(V,U_2) \cap B_2.\]

            \item By \cref{lemma_return_time_algebra_one} and \cref{lemma_return_to_intersection} and the definition of $B_2$,
            \[t_2 \in R(x,U_2) \cap B_2 = s_1^{-1} R(x,V) \cap s_1^{-1} B \cap s_2^{-1} R(x,V) \cap s_2^{-1} B.\]
            Therefore, $s_1 t_2, s_2 t_2 \in R(x,V) \cap B$.  Therefore
            \begin{align}
                \label{eqn_delta_it_two}
                s_3 \defeq s_1s_2t_2 \in s_1 (R(x,V) \cap B) \cap s_2 (R(x,V) \cap B).
            \end{align}

            \item Since $t_2 \in R(V,U_2)$, $V \cap t_2^{-1} U_2 \neq \emptyset$.  Since $s_1s_2$ acts surjectively (\cref{lemma_minimal_comm_action_implies_surjective}),
            \begin{align*}
                U_3 &\defeq (s_1s_2)^{-1} V \cap (s_1 s_3)^{-1} V \cap (s_2 s_3)^{-1} V \\
                &= (s_1s_2)^{-1} \big( V \cap (s_1 t_2)^{-1} V \cap (s_2 t_2)^{-1} V \big)\\
                &= (s_1s_2)^{-1} \big( V \cap t_2^{-1} U_2\big) \neq \emptyset.
            \end{align*}
            Note that by \cref{lemma_return_time_algebra_one} and \cref{lemma_return_to_intersection},
            \[R(x,U_3) = (s_1s_2)^{-1} R(x,V) \cap (s_1s_3)^{-1} R(x,V) \cap (s_2s_3)^{-1}R(x,V).\]
            Put
            \[B_3 \defeq (s_1s_2)^{-1} B \cap (s_1s_3)^{-1} B \cap (s_2s_3)^{-1} B.\]
        \end{enumerate}

        \item[Step 3]
        \begin{enumerate}
            \item By \cref{thm_delta_set_builder_prestep}, the set $R(x,U_3) \cap R(V,U_3)$ is syndetic.  By \cref{lemma_preimage_intersection_of_thick_is_thick}, the set $B_3$ is thick. By \cref{lemma_synd_thick_dual}, the intersection of a thick set and a syndetic set is non-empty.  Therefore, there exists
            \[t_3 \in R(x,U_3) \cap R(V,U_3) \cap B_3.\]

            \item By \cref{lemma_return_time_algebra_one} and \cref{lemma_return_to_intersection} and the definition of $B_3$,
            \begin{align*}
                t_3 &\in R(x,U_3) \cap B_3 \\
                &= (s_1s_2)^{-1} (R(x,V) \cap B) \cap (s_1s_3)^{-1} (R(x,V) \cap B) \cap (s_2s_3)^{-1}(R(x,V) \cap B).
            \end{align*}
            Therefore, $s_1 s_2 t_3, s_1 s_3 t_3, s_2 s_3 t_3 \in R(x,V) \cap B$.  Therefore
            \begin{align}
                \label{eqn_delta_it_three}
                s_4 \defeq s_1s_2 s_3t_3  \in s_1 (R(x,V) \cap B) \cap s_2 (R(x,V) \cap B) \cap s_3 (R(x,V) \cap B).
            \end{align}

            \item Since $t_3 \in R(V,U_3)$, $V \cap t_3^{-1} U_3 \neq \emptyset$.  Since $s_1s_2s_3$ acts surjectively (\cref{lemma_minimal_comm_action_implies_surjective}),
            \begin{align*}
                U_4 &\defeq (s_1s_2s_3)^{-1} V \cap (s_1 s_2 s_4)^{-1} V \cap (s_1 s_3 s_4)^{-1} V \cap (s_2s_3s_4)^{-1} V \\
                &= (s_1s_2s_3)^{-1} \big( V \cap (s_1 s_2t_3)^{-1} V \cap (s_1 s_3 t_3)^{-1} V \cap (s_2s_3 t_3)^{-1}V \big)\\
                &= (s_1s_2s_3)^{-1} \big( V \cap t_3^{-1} U_3\big) \neq \emptyset.
            \end{align*}
            Note that by \cref{lemma_return_time_algebra_one} and \cref{lemma_return_to_intersection},
            \[R(x,U_4) = (s_1s_2s_3)^{-1} R(x,V) \cap (s_1s_2s_4)^{-1} R(x,V) \cap (s_1s_3s_4)^{-1}R(x,V) \cap (s_2s_3s_4)^{-1}R(x,V).\]
            Put
            \[B_4 \defeq (s_1s_2s_3)^{-1} B \cap (s_1s_2s_4)^{-1} B \cap (s_1s_3s_4)^{-1} B \cap (s_2s_3s_4)^{-1} B.\]
        \end{enumerate}
    \end{enumerate}
    
    Iterate ad infinitum: we get $s_1, s_2, \dots$ such that for all $m < n$, $s_n \in s_m (R(x,V) \cap B)$, as was to be shown.  The first instances of this are shown in \eqref{eqn_delta_it_one}, \eqref{eqn_delta_it_two}, and \eqref{eqn_delta_it_three}.
\end{proof}

\subsection{Equivalences for dynamically syndetic sets}

A version of the following theorem can be found (for $\Z$-actions, or more generally?) in \cite{huang_shao_ye_2016}.

\begin{theorem}
\label{thm_dcthickbohr_is_delta_for_rxu}
    Let $G$ be a commutative semigroup, and let $B \subseteq G$ be CRTURP.  If $B \in \dcthickbohr$, then for all thick $H \subseteq G$, the set $R(x,U) \cap H \in \Delta$.
\end{theorem}

\begin{proof}
\uses{lemma_pr_family_concentration,lemma_dcthickbohr_is_pr,lemma_dcsyndeticbohr_dcthickbohr_dual,thm_rp_equals_s,thm_delta_from_kronecker}
    Since $B$ is a CRTURP set, there exists $X$ be a minimal $G$-system, $x \in X$, and $U \subseteq X$ clopen so that $B = R(x,U)$.  Assume $B \in \dcthickbohr$.

    Since $R(x,U) \in \dcthickbohr$, by \cref{lemma_pr_family_concentration}, \cref{lemma_dcthickbohr_is_pr}, and the fact that $U$ is closed, there exists $y \in U$ such that for all non-empty, open $U' \ni y$, the set $R(x,U') \in \dcthickbohr$.
    
    Let $\pi: X \to K$ be any equicontinuous factor of $X$.  We will show that $\pi x = \pi y$ by showing that for all open neighborhoods $V$ of $\pi x$ and $W$ of $\pi y$, $V \cap W \neq \emptyset$.  Let $V$ and $W$ be open neighborhoods of $\pi x$ and $\pi y$ respectively.  Since $K$ is equicontinuous and $V \ni \pi x$, $R(\pi x, V) \in \dcsyndeticbohr$. Since $\pi^{-1} W \ni y$ and $R(\pi x, V) \in \dcsyndeticbohr$, we have by \cref{lemma_dcsyndeticbohr_dcthickbohr_dual} that $R(x,\pi^{-1} W) \cap R(\pi x, V) \neq \emptyset$.  Therefore, there exists $g \in G$ such that $gx \in \pi^{-1} W$ and $g \pi x \in V$.  Thus, $\pi g x = g \pi x \in W$ and $g \pi x \in V$, so $V \cap W \neq \emptyset$, as desired.

    We conclude by showing that for all thick $H \subseteq G$, the set $R(x,U) \cap H \in \Delta$. Let $\pi: X \to K$ be the maximal equicontinuous factor of $X$.  We showed in the previous paragraph that $y \in U$ is such that $\pi x = \pi y$.  Therefore, $(x,y) \in \seq$.  By \cref{thm_rp_equals_s}, $\seq = \RP$, so $(x,y) \in \RP$.  Therefore, by \cref{thm_delta_from_kronecker}, $R(x,U) \cap H \in \Delta$.
\end{proof}

\subsection{Proof of \cref{thm_bfw_dual_abelian} and some corollaries}

If you only care about the combinatorics, the following is a weaker, but purely combinatorial and still highly-nontrivial-looking consequence:
\[\syndetic \classcapdual \Delta = \syndetic \classcapdual \Delta_0.\]

\begin{theorem}
\label{thm_bfw_dual_abelian}
    In a commutative semigroup $S$,
    \[\syndetic \classcapdual \Delta = \syndetic \classcapdual \Delta_0 = \syndetic \classcapdual \dcthickbohr.\]
\end{theorem}

\begin{proof}
\uses{thm_delta_in_deltazero,thm_delta_zero_set_of_bohr_rec,thm_classcapdual_is_monotonic,thm_lspwf_upgrade,thm_classcapdual_contained_in_intersection,thm_dcthickbohr_is_delta_for_rxu}
    That
    \[\syndetic \classcapdual \Delta \subseteq \syndetic \classcapdual \Delta_0 \subseteq \syndetic \classcapdual \dcthickbohr.\]
    follows from $\Delta \subseteq \Delta_0$ (\cref{thm_delta_in_deltazero}), $\Delta_0 \subseteq \dcthickbohr$ (\cref{thm_delta_zero_set_of_bohr_rec}), and the monotonicity of $\classcapdual$ (\cref{thm_classcapdual_is_monotonic}). So we have only left to show that
    \[\syndetic \classcapdual \dcthickbohr \subseteq \syndetic \classcapdual \Delta.\]
    According to \cref{thm_lspwf_upgrade}, it suffices to show this inclusion for CRTURP sets.  Let $B \in \syndetic \classcapdual \dcthickbohr$ be a CRTURP set.  From \cref{thm_classcapdual_contained_in_intersection}, we have that
    \[\syndetic \classcapdual \dcthickbohr \subseteq \dcthickbohr,\]
    whence $B \in \dcthickbohr$.  We have from \cref{thm_dcthickbohr_is_delta_for_rxu} that since $B \in \dcthickbohr$, $B \in \syndetic \classcapdual \Delta$, as desired.
\end{proof}

\begin{theorem}
\label{thm_delta_star_locally_bohr}
    Let $S$ be a commutative semigroup. Delta star sets are locally Bohr.  Actually:
    \[\Delta^* \subseteq \syndetic \classcapdual (\thick \classcap \dcsyndeticbohr).\]
\end{theorem}

\begin{proof}
\uses{thm_bfw_dual_abelian,lemma_synd_thick_dual,lemma_dcsyndeticbohr_dcthickbohr_dual,thm_combo_algebra_statement_one}
    From \cref{thm_bfw_dual_abelian}, we have that
    \[\syndetic \classcapdual \dcthickbohr \subseteq \syndetic \classcapdual \Delta.\]
    Since $\syndetic^* = \thick$ (\cref{lemma_synd_thick_dual}) and $\dcthickbohr^* = \dcsyndeticbohr$ (\cref{lemma_dcsyndeticbohr_dcthickbohr_dual}), we have the conclusion from \cref{thm_combo_algebra_statement_one}.
\end{proof}
